"""
backend/app/graph.py

LangGraph-powered Autonomous Multi-Agent Framework for PageIndex RAG.

Key Capabilities:
1. Decomposes complex/cross-document queries into precise sub-queries.
2. Parallel context retrieval using Tree-Walking & Corpus Narrowing,
   scoped to a single upload-batch (batch_id).
3. Reranks & evaluates context completeness dynamically.
4. Auto-correcting feedback loop for missing context (Max Hops control).
5. Synthesizes 99%+ accurate answers with complete source attribution/citations.
"""

import asyncio
import logging
import json
import re
from typing import List, Dict, Any, Optional, TypedDict

from sqlalchemy.orm import Session
from langgraph.graph import StateGraph, END

# Import existing core services from your codebase
from app.models.models import Document, TreeNode
from app.services.corpus_service import narrow_candidate_documents
from app.services.retrieval_agent import collect_relevant_nodes
from app.services.reranker_service import rerank_nodes
from app.services.ollama_service import chat_completion

logger = logging.getLogger("pageindex.langgraph")

# Configuration Constants
MAX_TOTAL_CONTEXT_NODES = 10
CONTEXT_CHARS_PER_NODE = 3000
MAX_HOPS = 3

# Prompts
DECOMPOSE_SYSTEM_PROMPT = (
    "You are an expert query analyzer. Your job is to break down a user query into concise, independent "
    "sub-queries required to search across a document database. If the query asks for comparisons or multi-part "
    "information, break it into individual sub-questions. If the query is simple, reply with just the original query.\n"
    "Respond ONLY with a JSON array of string queries. Example:\n"
    '["What is Company X revenue for Q3?", "What is Company Y revenue for Q3?"]'
)

EVALUATE_SYSTEM_PROMPT = (
    "You are an objective auditor evaluating retrieved document excerpts against a user query.\n"
    "Reply in EXACTLY ONE of these formats:\n"
    "COMPLETE\n"
    "or\n"
    "INCOMPLETE: <short description of missing information to search next>\n\n"
    "Only mark INCOMPLETE if crucial facts requested by the user are missing."
)

SYNTHESIS_SYSTEM_PROMPT = (
    "You are an elite research analyst. Synthesize a complete, structured, and 100% factual answer to the "
    "user query using ONLY the provided document excerpts. Explicitly attribute facts to their source document filename.\n"
    "If the query involves comparisons or tables, format the answer cleanly using Markdown tables.\n"
    "Do not extrapolate or invent facts beyond the excerpts."
)


# ==========================================
# 1. GRAPH STATE DEFINITION
# ==========================================
class RAGAgentState(TypedDict):
    db: Session
    batch_id: str                       # required - which upload-batch this chat is scoped to
    document_id: Optional[str]          # populated only for legacy single-document mode
    original_query: str
    sub_queries: List[str]
    accumulated_context: List[Dict[str, Any]]  # Stores {"document": Doc, "node": Node}
    reranked_context: List[Dict[str, Any]]
    missing_info_query: Optional[str]
    hop_count: int
    is_complete: bool
    final_answer: str
    sources: List[Dict[str, Any]]


# ==========================================
# 2. HELPER FUNCTIONS
# ==========================================
def _fetch_nodes_by_document(db: Session, document_ids: List[str]) -> Dict[str, List[TreeNode]]:
    if not document_ids:
        return {}
    all_nodes = db.query(TreeNode).filter(TreeNode.document_id.in_(document_ids)).all()
    by_doc: Dict[str, List[TreeNode]] = {doc_id: [] for doc_id in document_ids}
    for node in all_nodes:
        by_doc.setdefault(node.document_id, []).append(node)
    return by_doc


async def _collect_for_doc(db: Session, doc: Document, query: str, nodes: List[TreeNode]):
    if not nodes:
        return doc, []
    collected = await collect_relevant_nodes(db, doc.id, query, nodes=nodes)
    return doc, collected


def _build_context_text(nodes_with_docs: List[Dict[str, Any]]) -> tuple[str, List[Dict[str, Any]]]:
    context_blocks = []
    sources = []
    for item in nodes_with_docs:
        doc: Document = item["document"]
        node: TreeNode = item["node"]
        text = (node.raw_text or "").strip() or "(no direct text)"
        context_blocks.append(
            f"### [{doc.filename}] {node.title} (pages {node.page_start}-{node.page_end})\n"
            f"{text[:CONTEXT_CHARS_PER_NODE]}"
        )
        sources.append({
            "document_id": doc.id,
            "filename": doc.filename,
            "node_id": node.id,
            "title": node.title,
            "page_start": node.page_start,
            "page_end": node.page_end,
        })
    return "\n\n".join(context_blocks), sources


# ==========================================
# 3. GRAPH NODE FUNCTIONS
# ==========================================
async def decompose_query_node(state: RAGAgentState) -> Dict[str, Any]:
    """Node 1: Breaks complex query into sub-queries."""
    query = state["original_query"]
    logger.info("--- [LangGraph] Decomposing Query (batch=%s): %r ---", state.get("batch_id"), query)

    # If scoped to a single document, skip heavy multi-query decomposition
    if state.get("document_id"):
        return {"sub_queries": [query], "hop_count": 0, "accumulated_context": []}

    prompt = f"User Query: {query}"
    response = await chat_completion(prompt=prompt, system=DECOMPOSE_SYSTEM_PROMPT)

    try:
        match = re.search(r"\[.*\]", response, re.DOTALL)
        if match:
            sub_queries = json.loads(match.group(0))
        else:
            sub_queries = [query]
    except Exception as e:
        logger.warning("Failed to parse sub-queries JSON: %s. Using original query.", e)
        sub_queries = [query]

    logger.info("Generated Sub-Queries: %s", sub_queries)
    return {"sub_queries": sub_queries, "hop_count": 0, "accumulated_context": []}


async def parallel_retrieval_node(state: RAGAgentState) -> Dict[str, Any]:
    """Node 2: Executes parallel tree retrieval across all active sub-queries."""
    db = state["db"]
    doc_id = state.get("document_id")
    batch_id = state["batch_id"]
    sub_queries = state.get("sub_queries", [state["original_query"]])
    current_hop = state.get("hop_count", 0) + 1

    logger.info("--- [LangGraph] Retrieval Pass Hop %d/%d (batch=%s) ---", current_hop, MAX_HOPS, batch_id)

    new_gathered: List[Dict[str, Any]] = []

    if doc_id:
        # Legacy Single Document Mode
        doc = db.query(Document).filter(Document.id == doc_id).first()
        nodes = db.query(TreeNode).filter(TreeNode.document_id == doc_id).all()
        for sq in sub_queries:
            _, collected = await _collect_for_doc(db, doc, sq, nodes)
            for n in collected:
                new_gathered.append({"document": doc, "node": n})
    else:
        # Batch-Scoped Multi-Document Mode - only documents belonging to
        # batch_id are ever considered as candidates.
        for sq in sub_queries:
            candidates = await narrow_candidate_documents(db, sq, batch_id=batch_id)
            if not candidates:
                continue
            cand_ids = [c.id for c in candidates]
            nodes_by_doc = _fetch_nodes_by_document(db, cand_ids)

            tasks = [_collect_for_doc(db, c, sq, nodes_by_doc.get(c.id, [])) for c in candidates]
            results = await asyncio.gather(*tasks)

            for d, collected_nodes in results:
                for n in collected_nodes:
                    new_gathered.append({"document": d, "node": n})

    # Deduplicate accumulated context by node.id
    existing_items = state.get("accumulated_context", [])
    seen_node_ids = {item["node"].id for item in existing_items}

    merged = list(existing_items)
    for item in new_gathered:
        if item["node"].id not in seen_node_ids:
            seen_node_ids.add(item["node"].id)
            merged.append(item)

    logger.info("Total accumulated context nodes after Hop %d: %d", current_hop, len(merged))
    return {"accumulated_context": merged, "hop_count": current_hop}


async def evaluate_and_rerank_node(state: RAGAgentState) -> Dict[str, Any]:
    """Node 3: Reranks gathered context using embeddings & audits completeness."""
    accumulated = state.get("accumulated_context", [])
    query = state["original_query"]
    hop_count = state.get("hop_count", 1)

    if not accumulated:
        logger.warning("No context gathered yet.")
        return {"is_complete": False, "missing_info_query": query, "reranked_context": []}

    # Transform for reranker service format: List[tuple[Document, TreeNode]]
    raw_pairs = [(item["document"], item["node"]) for item in accumulated]
    reranked_pairs = await rerank_nodes(query, raw_pairs, top_n=MAX_TOTAL_CONTEXT_NODES)

    reranked_items = [{"document": d, "node": n} for d, n in reranked_pairs]
    context_text, _ = _build_context_text(reranked_items)

    # Stop condition check if max hops reached
    if hop_count >= MAX_HOPS:
        logger.info("Max hops (%d) reached. Proceeding to answer.", MAX_HOPS)
        return {"is_complete": True, "reranked_context": reranked_items, "missing_info_query": None}

    # Completeness audit
    eval_prompt = f"User Question: {query}\n\nGathered Excerpts:\n{context_text}\n\nIs this sufficient?"
    response = await chat_completion(prompt=eval_prompt, system=EVALUATE_SYSTEM_PROMPT)

    stripped = response.strip()
    match = re.match(r"INCOMPLETE\s*:\s*(.+)", stripped, re.IGNORECASE | re.DOTALL)

    if match:
        missing_query = match.group(1).strip()
        logger.info("Evaluation: INCOMPLETE. Gap identified: %r", missing_query)
        return {
            "is_complete": False,
            "missing_info_query": missing_query,
            "sub_queries": [missing_query],  # Overwrite sub-queries for next hop loop
            "reranked_context": reranked_items,
        }

    logger.info("Evaluation: COMPLETE context verified.")
    return {"is_complete": True, "missing_info_query": None, "reranked_context": reranked_items}


async def synthesize_answer_node(state: RAGAgentState) -> Dict[str, Any]:
    """Node 4: Synthesizes final response with verified citations."""
    reranked = state.get("reranked_context", [])
    query = state["original_query"]

    if not reranked:
        return {
            "final_answer": "I could not find relevant information in the uploaded documents to answer your query.",
            "sources": [],
        }

    context_text, sources = _build_context_text(reranked)
    final_prompt = f"Document Excerpts:\n\n{context_text}\n\nUser Question: {query}"

    logger.info("Synthesizing final response from %d context nodes...", len(reranked))
    final_answer = await chat_completion(prompt=final_prompt, system=SYNTHESIS_SYSTEM_PROMPT)

    return {"final_answer": final_answer, "sources": sources}


# ==========================================
# 4. CONDITIONAL ROUTER
# ==========================================
def completion_router(state: RAGAgentState) -> str:
    """Routes execution: If complete or max hops hit -> Synthesize, else Loop Back."""
    if state.get("is_complete") or state.get("hop_count", 0) >= MAX_HOPS:
        return "synthesize"
    return "retrieve"


# ==========================================
# 5. GRAPH BUILDER & EXECUTOR
# ==========================================
def build_rag_graph() -> StateGraph:
    workflow = StateGraph(RAGAgentState)

    # Add Nodes
    workflow.add_node("decompose", decompose_query_node)
    workflow.add_node("retrieve", parallel_retrieval_node)
    workflow.add_node("evaluate", evaluate_and_rerank_node)
    workflow.add_node("synthesize", synthesize_answer_node)

    # Set Entry Point
    workflow.set_entry_point("decompose")

    # Connect Edges
    workflow.add_edge("decompose", "retrieve")
    workflow.add_edge("retrieve", "evaluate")

    # Conditional Loop Edge
    workflow.add_conditional_edges(
        "evaluate",
        completion_router,
        {
            "synthesize": "synthesize",
            "retrieve": "retrieve",
        },
    )

    workflow.add_edge("synthesize", END)
    return workflow.compile()


# Global compiled instance
langgraph_app = build_rag_graph()


# Interface function to trigger graph execution
async def run_langgraph_chat(
    db: Session,
    query: str,
    batch_id: str,
    document_id: Optional[str] = None,
) -> Dict[str, Any]:
    """
    Main entry point for API routers to execute LangGraph agent execution.

    batch_id is now required: every chat is scoped to one upload-batch, and
    cross-document retrieval only ever searches within that batch's
    documents. document_id is kept as an optional override for legacy
    single-document scoping and, when provided, must belong to batch_id.
    """
    initial_state: RAGAgentState = {
        "db": db,
        "batch_id": batch_id,
        "document_id": document_id,
        "original_query": query,
        "sub_queries": [],
        "accumulated_context": [],
        "reranked_context": [],
        "missing_info_query": None,
        "hop_count": 0,
        "is_complete": False,
        "final_answer": "",
        "sources": [],
    }

    result_state = await langgraph_app.ainvoke(initial_state)

    return {
        "answer": result_state.get("final_answer"),
        "sources": result_state.get("sources", []),
        "hops_used": result_state.get("hop_count", 1),
    }

def _build_batch_out(batch: Batch) -> BatchOut:
    return BatchOut(
        id=batch.id,
        name=batch.name,
        created_at=batch.created_at,
        documents=[_build_document_out(d) for d in batch.documents],
    )


@router.post("/upload", response_model=BatchOut)
async def upload_batch(
    files: List[UploadFile] = File(...),
    db: Session = Depends(get_db),
):
    """
    Accepts one or more PDFs in a single request, creates one Batch row,
    and creates a Document row per file under that batch. Every document
    is then enqueued for indexing - the indexing queue already processes
    strictly one document at a time (see indexing_queue.py), so a 200-file
    batch is safe to submit in one call; it just takes a while to finish.
    """
    if not files:
        raise HTTPException(status_code=400, detail="No files provided.")

    for f in files:
        if not f.filename.lower().endswith(".pdf"):
            raise HTTPException(status_code=400, detail=f"Only PDF files are supported ({f.filename}).")

    os.makedirs(settings.upload_dir, exist_ok=True)

    batch_name = files[0].filename if len(files) == 1 else f"Batch of {len(files)} documents"
    batch = Batch(name=batch_name)
    db.add(batch)
    db.commit()
    db.refresh(batch)

    created_docs = []
    for f in files:
        stored_name = f"{uuid.uuid4()}_{f.filename}"
        dest_path = os.path.join(settings.upload_dir, stored_name)

        with open(dest_path, "wb") as out_file:
            shutil.copyfileobj(f.file, out_file)

        doc = Document(
            batch_id=batch.id,
            filename=f.filename,
            file_path=dest_path,
            status=DocumentStatus.uploaded,
        )
        db.add(doc)
        created_docs.append(doc)

    db.commit()

    for doc in created_docs:
        db.refresh(doc)
        queue_position = enqueue_document(doc.id)
        logger.info(
            "Document %s (batch %s) enqueued for indexing (queue size: %d)",
            doc.id, batch.id, queue_position,
        )

    db.refresh(batch)
    return _build_batch_out(batch)


@router.get("", response_model=List[BatchOut])
def list_batches(db: Session = Depends(get_db)):
    batches = db.query(Batch).order_by(Batch.created_at.desc()).all()
    return [_build_batch_out(b) for b in batches]

    if not batch:
        raise HTTPException(status_code=404, detail="Batch not found")
    return _build_batch_out(batch)

def _build_message_out(db: Session, message: Message) -> MessageOut:
    """Enriches a stored Message with the actual filenames/titles/pages of
    any cited TreeNodes, since cited_node_ids in the DB is just a JSON
    list of ids. Filename is included because a batch answer can cite
    sections from several different documents at once."""
    cited_sections: list[CitedSection] = []
    if message.cited_node_ids:
        try:
            ids = json.loads(message.cited_node_ids)
        except (TypeError, ValueError):
            ids = []
        if ids:
            nodes = db.query(TreeNode).filter(TreeNode.id.in_(ids)).all()
            by_id = {n.id: n for n in nodes}

            doc_ids = {n.document_id for n in nodes}
            docs_by_id = {
                d.id: d for d in db.query(Document).filter(Document.id.in_(doc_ids)).all()
            } if doc_ids else {}

            for node_id in ids:
                n = by_id.get(node_id)
                if n:
                    doc = docs_by_id.get(n.document_id)
                    cited_sections.append(
                        CitedSection(
                            id=n.id,
                            filename=doc.filename if doc else "unknown",
                            title=n.title,
                            page_start=n.page_start,
                            page_end=n.page_end,
                        )
                    )

    return MessageOut(
        id=message.id,
        role=message.role.value if hasattr(message.role, "value") else message.role,
        content=message.content,
        created_at=message.created_at,
        cited_sections=cited_sections,
    )

@router.post("/sessions/{session_id}/messages", response_model=MessageOut)
async def send_message(session_id: str, body: MessageIn, db: Session = Depends(get_db)):
    session = db.query(ChatSession).filter(ChatSession.id == session_id).first()
    if not session:
        raise HTTPException(status_code=404, detail="Chat session not found")

    user_msg = Message(session_id=session_id, role=MessageRole.user, content=body.content)
    db.add(user_msg)
    db.commit()

    documents = db.query(Document).filter(Document.batch_id == session.batch_id).all()
    not_ready = [d for d in documents if d.status != DocumentStatus.ready]

    if not documents or not_ready:
        if not documents:
            wait_message = "This batch has no documents."
        else:
            statuses = ", ".join(sorted({d.status.value for d in not_ready}))
            wait_message = (
                f"This batch is still being processed ({len(not_ready)} of {len(documents)} "
                f"document(s) still in progress: {statuses}). Please wait for indexing to "
                "finish on all documents before asking questions - answers could otherwise "
                "miss information from the documents that aren't ready yet."
            )
        assistant_msg = Message(session_id=session_id, role=MessageRole.assistant, content=wait_message)
        db.add(assistant_msg)
        db.commit()
        db.refresh(assistant_msg)
        return _build_message_out(db, assistant_msg)

    # Invoke LangGraph engine (batch-scoped, cross-document retrieval) or
    # fall back seamlessly to the plain async multi-doc orchestrator.
    if HAS_LANGGRAPH:
        logger.info("Processing message via LangGraph state machine agent (batch=%s)...", session.batch_id)
        graph_result = await run_langgraph_chat(
            db=db,
            query=body.content,
            batch_id=session.batch_id,
        )
        answer_text = graph_result.get("answer", "")
        cited_ids = [s["node_id"] for s in graph_result.get("sources", []) if "node_id" in s]
    else:
        logger.info("Processing message via multi-doc orchestrator (batch=%s)...", session.batch_id)
        result = await answer_question_multi_doc(db, body.content, batch_id=session.batch_id)
        answer_text = result["answer"]
        cited_ids = [s["node_id"] for s in result.get("sources", []) if "node_id" in s]

    assistant_msg = Message(
        session_id=session_id,
        role=MessageRole.assistant,
        content=answer_text,
        cited_node_ids=json.dumps(cited_ids),
    )
    db.add(assistant_msg)
    db.commit()
    db.refresh(assistant_msg)

    return _build_message_out(db, assistant_msg)

        raise HTTPException(status_code=404, detail="Document not found")

    rows = (
        db.query(TreeNode)
        .filter(TreeNode.document_id == document_id)
        .order_by(TreeNode.order_index)
        .all()
    )

        row.id: TreeNodeOut(
            id=row.id,
            parent_id=row.parent_id,
            title=row.title,
            summary=row.summary,
            keywords=_parse_keywords(row.keywords),
            page_start=row.page_start,
            page_end=row.page_end,
            children=[],
        )
        for row in rows

    for row in rows:
        node_out = by_id[row.id]
        if row.parent_id and row.parent_id in by_id:
            by_id[row.parent_id].children.append(node_out)
        else:
            roots.append(node_out)

    return roots

### backend/app/services/corpus_service.py
"""
Phase 6 (Chunk 3), Step 2 (upgraded for 200-document scale, then again for
batch scoping): corpus-wide candidate narrowing.

When a user asks a question in "search across the batch" mode, we can't
afford to run the full tree-search (collect_relevant_nodes) on every
document in the batch - that's one or more Ollama calls PER DOCUMENT PER
TREE LEVEL, which doesn't scale past a handful of files.

Instead, this module does a cheap pass first: show the LLM every
candidate document's rollup summary + keywords (generated in Chunk 2 by
summary_service.py) and ask which documents are even worth searching for
this question. Only those "candidate" documents go on to the real
per-document tree-search in the next step (multi-doc orchestrator).

IMPORTANT: narrowing is now scoped to a single batch_id. A batch is the
set of documents a user uploaded together and is chatting with as one
unit - documents from OTHER batches must never leak into this search,
even if they're both "ready". Without this filter, a 10-document batch
would get its candidates diluted (or replaced) by unrelated documents
from a different 200-document batch sitting in the same table.

Only documents with status == ready are considered, since summary/
keywords/tree nodes don't exist yet for documents still indexing.
"""
import asyncio
import json
import logging
import re

from sqlalchemy.orm import Session

from app.models.models import Document, DocumentStatus
from app.services.ollama_service import chat_completion

logger = logging.getLogger("pageindex.corpus")

# How many candidate documents to hand off to the per-document tree-search
# (Step 4 in the pipeline). 10-20 chosen so that even if a couple of true
# positives get missed by the LLM narrowing pass, there's enough headroom
# for the answer to still be found among the survivors.
MAX_CANDIDATE_DOCUMENTS = 15

# How many documents go into ONE narrowing prompt. Keeps each individual
# LLM call's listing short enough for a 4B model to reason over reliably.
# e.g. 200 documents / 40 per batch = 5 LLM calls, run in parallel.
DOCUMENTS_PER_BATCH = 40

CANDIDATE_SELECTION_SYSTEM_PROMPT = (
    "You are choosing which documents are worth searching to answer a "
    "question. You are shown a numbered list of documents, each with a "
    "short summary and keywords. Reply with ONLY the numbers of documents "
    "that might contain relevant information, separated by commas "
    "(example: 1,3,4). Include a document even if you're only somewhat "
    "sure it's relevant - it's fine to include more than the minimum. "
    "If truly none of them seem relevant, reply with exactly: NONE. "
    "Do not add any explanation."
)


def _parse_keywords(raw: str | None) -> list[str]:
    if not raw:
        return []
    try:
        return json.loads(raw)
    except (TypeError, ValueError):
        return []


def _format_document_listing(documents: list[Document]) -> str:
    lines = []
    for i, doc in enumerate(documents):
        keywords = _parse_keywords(doc.keywords)
        keywords_str = ", ".join(keywords) if keywords else "(none)"
        summary = doc.summary or "(no summary available)"
        lines.append(f"{i + 1}. {doc.filename}\n   Summary: {summary}\n   Keywords: {keywords_str}")
    return "\n".join(lines)


def _chunk_list(items: list, size: int) -> list[list]:
    return [items[i:i + size] for i in range(0, len(items), size)]


async def _narrow_one_batch(question: str, batch: list[Document], batch_label: str) -> list[Document]:
    """
    Runs the narrowing prompt on a single LLM-call-sized batch of documents
    (<= DOCUMENTS_PER_BATCH) and returns the Document rows from THIS batch
    that the LLM picked.

    Note: this is an internal chunking batch for the LLM prompt, distinct
    from the user-facing upload Batch (models.Batch) - a single upload
    Batch of e.g. 200 documents gets split into several of these here.
    """
    listing = _format_document_listing(batch)
    prompt = f"Question: {question}\n\nDocuments:\n{listing}\n\nWhich document numbers might contain the answer?"

    logger.info(
        "[%s] Asking LLM to narrow among %d documents: %s",
        batch_label, len(batch), [d.filename for d in batch],
    )

    response = await chat_completion(prompt=prompt, system=CANDIDATE_SELECTION_SYSTEM_PROMPT)

    numbers = sorted(set(int(x) for x in re.findall(r"\d+", response)))
    selected = [batch[n - 1] for n in numbers if 1 <= n <= len(batch)]

    logger.info(
        "[%s] LLM raw reply=%r -> parsed_numbers=%s -> picked=%s",
        batch_label, response, numbers, [d.filename for d in selected],
    )

    return selected


async def narrow_candidate_documents(
    db: Session,
    question: str,
    batch_id: str,
    max_candidates: int = MAX_CANDIDATE_DOCUMENTS,
    documents_per_batch: int = DOCUMENTS_PER_BATCH,
) -> list[Document]:
    """
    Returns a shortlist of Document rows worth running the full tree-search
    on for this question, using only their rollup summary+keywords.

    Scoped to a single upload-batch (batch_id) - documents from any other
    batch are never considered, no matter how many total documents exist
    in the system.

    For batches larger than `documents_per_batch`, documents are split
    into LLM-call-sized chunks and narrowed in parallel (one LLM call per
    chunk, not one call for the whole batch) - this keeps each prompt
    short and lets narrowing scale to hundreds of documents without the
    accuracy hit of one giant list.
    """
    documents = (
        db.query(Document)
        .filter(Document.batch_id == batch_id, Document.status == DocumentStatus.ready)
        .order_by(Document.created_at.desc())
        .all()
    )

    if not documents:
        logger.info("No ready documents in batch %s - nothing to narrow", batch_id)
        return []

    if len(documents) <= max_candidates:
        logger.info(
            "Batch %s has only %d ready documents (<= max_candidates=%d) - "
            "skipping LLM narrowing, using all of them",
            batch_id, len(documents), max_candidates,
        )
        return documents

    chunks = _chunk_list(documents, documents_per_batch)
    logger.info(
        "Narrowing %d documents in batch %s across %d chunk(s) of up to %d each (parallel)",
        len(documents), batch_id, len(chunks), documents_per_batch,
    )

    chunk_results = await asyncio.gather(*[
        _narrow_one_batch(question, chunk, batch_label=f"batch={batch_id} chunk={i + 1}/{len(chunks)}")
        for i, chunk in enumerate(chunks)
    ])

    # Merge picks from all chunks, preserving order and de-duplicating
    # (a document can only appear in one chunk anyway, but stay safe).
    seen_ids = set()
    merged: list[Document] = []
    for chunk_selected in chunk_results:
        for doc in chunk_selected:
            if doc.id not in seen_ids:
                seen_ids.add(doc.id)
                merged.append(doc)

    logger.info(
        "Merged candidates across all chunks for batch %s: %d documents -> %s",
        batch_id, len(merged), [d.filename for d in merged],
    )

    if not merged:
        logger.info(
            "LLM found no relevant documents in any chunk - falling back to "
            "most recent %d in this batch as a safety net", max_candidates,
        )
        return documents[:max_candidates]

    if len(merged) > max_candidates:
        logger.info(
            "Merged candidates (%d) exceed max_candidates (%d) - truncating "
            "to the %d most recently created among them",
            len(merged), max_candidates, max_candidates,
        )
        merged = merged[:max_candidates]

    return merged

### Indexing_queue.py

"""
Ensures documents are indexed strictly one at a time, no matter how many
uploads arrive close together (e.g. a 200-file batch upload). A single
dedicated worker thread pulls document ids off a FIFO queue and processes
them sequentially - this keeps Ollama from being hit with dozens of
concurrent indexing requests at once.
"""
import logging
import queue
import threading

from app.services.indexing_service import process_document

logger = logging.getLogger("pageindex.queue")

_queue: "queue.Queue[str]" = queue.Queue()
_worker_started = False
_lock = threading.Lock()


def _worker_loop() -> None:
    while True:
        document_id = _queue.get()
        try:
            logger.info(
                "Indexing queue: starting document %s (%d remaining after this)",
                document_id, _queue.qsize(),
            )
            process_document(document_id)
        except Exception:
            logger.exception("Indexing queue: unhandled error processing document %s", document_id)
        finally:
            _queue.task_done()


def start_worker() -> None:
    """Call once at app startup. Safe to call multiple times."""
    global _worker_started
    with _lock:
        if not _worker_started:
            t = threading.Thread(target=_worker_loop, daemon=True, name="indexing-worker")
            t.start()
            _worker_started = True
            logger.info("Indexing worker thread started")


def enqueue_document(document_id: str) -> int:
    """Adds a document to the indexing queue. Returns the new queue size."""
    _queue.put(document_id)
    return _queue.qsize()

"""
Runs as a FastAPI background task right after upload. Creates its own DB
session since the request-scoped session from the upload endpoint is

"""
import traceback
from app.db.database import SessionLocal
from app.models.models import Document, DocumentStatus
from app.services.pdf_service import parse_pdf_to_tree
from app.services.tree_service import persist_tree
from app.services.summary_service import generate_summaries_for_document

def process_document(document_id: str) -> None:
    db = SessionLocal()
    try:
        doc = db.query(Document).filter(Document.id == document_id).first()
        if not doc:
            return

        doc.status = DocumentStatus.parsing
        db.commit()

        result = parse_pdf_to_tree(doc.file_path)

# multi_doc_service.py

"""Phase 6 (Chunk 3 + Chunk 4, then batch-scoping): multi-document orchestrator.

Chunk 3 built the base pipeline:
  - corpus_service.narrow_candidate_documents(): cheap pass that shortlists
    candidate documents out of a batch using rollup summary+keywords.
  - retrieval_agent.collect_relevant_nodes(): existing single-document
    tree-search, reused unchanged, run per candidate document in parallel.
  - reranker_service.rerank_nodes(): (Chunk 4, Step 4a) embedding-based
    cutoff replacing the old naive truncation.

Chunk 4, Step 4b adds a MULTI-HOP LOOP on top of that base pipeline:
some questions need evidence from more than one "round" of searching (e.g.
"compare Company X's and Company Y's Q3 revenue" - a single narrow+search
pass might surface only one company's numbers). After gathering context,
a cheap completeness-check LLM call decides if the context is enough to
answer the question. If not, it also tells us what's missing, we gather
context for THAT, merge it with what we already have, and try again -
up to MAX_HOPS times total, so the loop always terminates.

This orchestrator is now BATCH-SCOPED: every call requires a batch_id, and
every narrowing pass only considers documents belonging to that batch, so
a 10-document batch and a 200-document batch never bleed into each other
even though they live in the same documents table.

The single-hop gathering logic (narrow -> parallel tree-search -> rerank)
is pulled into _gather_context() so the multi-hop loop can call it
repeatedly without duplicating that logic.
"""
import asyncio
import logging
import re

from sqlalchemy.orm import Session

from app.models.models import Document, TreeNode
from app.services.corpus_service import narrow_candidate_documents
from app.services.retrieval_agent import collect_relevant_nodes
from app.services.reranker_service import rerank_nodes
from app.services.ollama_service import chat_completion

logger = logging.getLogger("pageindex.multidoc")

# How many nodes survive reranking to go into the final answer prompt.
MAX_TOTAL_CONTEXT_NODES = 8
CONTEXT_CHARS_PER_NODE = 3000

# Multi-hop loop bounds (Chunk 4, Step 4b). MAX_HOPS=3 means: 1 initial
# gather + up to 2 additional "fill the gap" gathers, matching the
# project plan's "max 2-3 hops". This is a hard cap so the loop always
# terminates even if the completeness-check keeps saying "incomplete".
MAX_HOPS = 3

MULTI_DOC_ANSWER_SYSTEM_PROMPT = (
    "Answer the user's question using ONLY the document excerpts provided "
    "below. The excerpts come from multiple different source documents, "

def _fetch_nodes_by_document(db: Session, document_ids: list[str]) -> dict[str, list[TreeNode]]:
    """One bulk query for all candidate documents' tree nodes, grouped by document_id."""
    if not document_ids:
        return {}

    all_nodes = (
        db.query(TreeNode)
        .filter(TreeNode.document_id.in_(document_ids))
        .all()
    )

    by_document: dict[str, list[TreeNode]] = {doc_id: [] for doc_id in document_ids}
    for node in all_nodes:
        by_document.setdefault(node.document_id, []).append(node)

    return by_document


async def _collect_for_document(
    db: Session,
    document: Document,
    search_query: str,
    nodes: list[TreeNode],
) -> tuple[Document, list[TreeNode]]:
    """Runs the existing single-document tree-search for one candidate document."""
    if not nodes:
        logger.info("Document %s (%s) has no tree nodes - skipping", document.id, document.filename)
        return document, []

    collected = await collect_relevant_nodes(db, document.id, search_query, nodes=nodes)
    return document, collected


async def _gather_context(
    db: Session,
    search_query: str,
    batch_id: str,
    hop_label: str,
) -> list[tuple[Document, TreeNode]]:
    """
    One full "round" of gathering: narrow candidate documents WITHIN
    batch_id for search_query, run tree-search on each candidate in
    parallel, and return the raw (document, node) pairs collected (NOT
    yet reranked - the caller merges across hops before doing a final
    rerank).
    """
    candidates = await narrow_candidate_documents(db, search_query, batch_id=batch_id)
    if not candidates:
        logger.info("[%s] No candidate documents found in batch %s for query %r", hop_label, batch_id, search_query)
        return []

    candidate_ids = [doc.id for doc in candidates]
    nodes_by_document = _fetch_nodes_by_document(db, candidate_ids)

    logger.info(
        "[%s] Running tree-search in parallel across %d candidate documents (batch=%s) for query %r: %s",
        hop_label, len(candidates), batch_id, search_query, [d.filename for d in candidates],
    )

    results = await asyncio.gather(*[
        _collect_for_document(db, doc, search_query, nodes_by_document.get(doc.id, []))
        for doc in candidates
    ])

    gathered: list[tuple[Document, TreeNode]] = []
    for document, collected_nodes in results:
        for node in collected_nodes:
            gathered.append((document, node))

    logger.info("[%s] Gathered %d raw nodes", hop_label, len(gathered))
    return gathered


def _merge_unique(
    existing: list[tuple[Document, TreeNode]],
    new: list[tuple[Document, TreeNode]],
) -> list[tuple[Document, TreeNode]]:
    """Merges two (document, node) lists, de-duplicating by node id."""
    seen_node_ids = {node.id for _, node in existing}
    merged = list(existing)
    for document, node in new:
        if node.id not in seen_node_ids:
            seen_node_ids.add(node.id)
            merged.append((document, node))
    return merged


async def _check_completeness(question: str, context_text: str) -> tuple[bool, str | None]:
    """
    Asks the LLM whether the gathered context is enough to answer the
    question. Returns (is_complete, missing_info_query_or_None).
    """
    prompt = f"Question: {question}\n\nGathered excerpts:\n{context_text}\n\nIs this enough to fully answer the question?"
    response = await chat_completion(prompt=prompt, system=COMPLETENESS_CHECK_SYSTEM_PROMPT)

    stripped = response.strip()
    match = re.match(r"INCOMPLETE\s*:\s*(.+)", stripped, re.IGNORECASE | re.DOTALL)
    if match:
        missing_query = match.group(1).strip()
        logger.info("Completeness check: INCOMPLETE - missing=%r", missing_query)
        return False, missing_query or None

    logger.info("Completeness check: COMPLETE (raw reply=%r)", stripped)
    return True, None


def _build_context_text(nodes_with_documents: list[tuple[Document, TreeNode]]) -> tuple[str, list[dict]]:
    context_blocks = []
    sources = []
    for document, node in nodes_with_documents:
        text = (node.raw_text or "").strip() or "(no direct text - see subsections)"
        context_blocks.append(
            f"### [{document.filename}] {node.title} (pages {node.page_start}-{node.page_end})\n"
            f"{text[:CONTEXT_CHARS_PER_NODE]}"
        )
        sources.append({
            "document_id": document.id,
            "filename": document.filename,
            "node_id": node.id,
            "title": node.title,
            "page_start": node.page_start,
            "page_end": node.page_end,
        })
    return "\n\n".join(context_blocks), sources


async def answer_question_multi_doc(db: Session, question: str, batch_id: str) -> dict:
    """
    Answers a question by searching across every ready document in
    batch_id, with a multi-hop loop (Chunk 4, Step 4b) that re-searches
    for missing information when the first pass's context looks
    incomplete.

    Returns:
        {
            "answer": str,
            "sources": [{"document_id", "filename", "node_id", "title",
                          "page_start", "page_end"}, ...],
            "hops_used": int,
        }
    """
    logger.info("=== New multi-document question for batch %s: %r ===", batch_id, question)

    accumulated: list[tuple[Document, TreeNode]] = []
    search_query = question

    for hop in range(1, MAX_HOPS + 1):
        hop_label = f"batch={batch_id} hop={hop}/{MAX_HOPS}"

        new_nodes = await _gather_context(db, search_query, batch_id=batch_id, hop_label=hop_label)
        accumulated = _merge_unique(accumulated, new_nodes)

        if not accumulated:
            logger.info("[%s] No relevant nodes found in any candidate document yet", hop_label)
            if hop == MAX_HOPS:
                break
            continue

        # Rerank the full accumulated set against the ORIGINAL question
        # (not the sub-query) so relevance is always judged against what
        # the user actually asked, even on later hops.
        reranked = await rerank_nodes(question, accumulated, top_n=MAX_TOTAL_CONTEXT_NODES)
        context_text, sources = _build_context_text(reranked)

        if hop == MAX_HOPS:
            logger.info("[%s] Reached MAX_HOPS - using accumulated context as-is", hop_label)
            break

        is_complete, missing_query = await _check_completeness(question, context_text)
        if is_complete or not missing_query:
            logger.info("[%s] Context judged complete - stopping hops", hop_label)
            break

        logger.info("[%s] Context incomplete - next hop will search for: %r", hop_label, missing_query)
        search_query = missing_query

    if not accumulated:
        logger.info("No relevant nodes found across any hop in batch %s", batch_id)
        answer = await chat_completion(
            prompt=f"Question: {question}\n\n(No relevant section of any document was found.)",
            system=NO_CONTEXT_SYSTEM_PROMPT,
        )
        return {"answer": answer, "sources": [], "hops_used": hop}

    reranked = await rerank_nodes(question, accumulated, top_n=MAX_TOTAL_CONTEXT_NODES)
    context_text, sources = _build_context_text(reranked)

    logger.info("Generating final multi-document answer from %d context nodes (batch=%s, hops_used=%d)", len(reranked), batch_id, hop)
    final_prompt = f"Document excerpts:\n\n{context_text}\n\nQuestion: {question}"
    answer = await chat_completion(prompt=final_prompt, system=MULTI_DOC_ANSWER_SYSTEM_PROMPT)

    return {"answer": answer, "sources": sources, "hops_used": hop}

async def get_embedding(text: str) -> list[float]:
    """
    Gets a vector embedding for a piece of text from Ollama's embedding
    endpoint (separate from /api/chat - this uses a dedicated embedding
    model, e.g. nomic-embed-text, not the chat model qwen3:4b).

    Used by reranker_service.py for cheap, LLM-free relevance scoring -
    no /api/chat call involved here, so this is fast even for many nodes.
    """
    payload = {
        "model": settings.ollama_embedding_model,
        "input": text,
    }

    start = time.monotonic()
    try:
        async with httpx.AsyncClient(timeout=settings.ollama_timeout_seconds) as client:
            resp = await client.post(f"{settings.ollama_base_url}/api/embed", json=payload)
            resp.raise_for_status()
            data = resp.json()
    except Exception:
        elapsed = time.monotonic() - start
        logger.exception("Ollama EMBED REQUEST FAILED after %.1fs", elapsed)
        raise

    elapsed = time.monotonic() - start
    # Ollama's /api/embed returns {"embeddings": [[...]]} - a list of vectors,
    # one per input. We send one input at a time, so take the first.
    embeddings = data.get("embeddings", [])
    if not embeddings:
        raise ValueError(f"Ollama returned no embeddings for input of {len(text)} chars")

    vector = embeddings[0]
    logger.info(
        "Ollama EMBED: elapsed=%.1fs text_chars=%d vector_dim=%d",
        elapsed, len(text), len(vector),
    )
    return vector


def _strip_thinking(content: str) -> str:
    """
    Some Ollama versions don't fully honor `"think": false` for reasoning
    models like qwen3, and still emit a hidden chain-of-thought ending in
    `</think>` before the real answer. Strip everything up to and
    including the last `</think>` so callers only see the final answer.
    """
    if "</think>" in content:
        content = content.split("</think>")[-1]
    return content.strip()

# pdf_service.py
"""
Phase 2: PDF parsing + structure detection.

This module is pure logic (no DB, no FastAPI) so it's easy to test on its
own. It takes a PDF file path and returns a nested tree of sections:

    {
      "page_count": int,
      "tree": [
        {
          "title": str, "summary": None, "raw_text": str,
          "page_start": int, "page_end": int,
          "children": [ ... same shape ... ]
        },
        ...
      ]
    }

`summary` is always None here - Phase 3 fills it in with per-node LLM
summaries. This module only figures out the document's structure.

IMPORTANT (added after observing degenerate trees on SEC filings like
10-K/10-Q): font-size-based heading detection can fail to find ANY
section headers inside a document body (e.g. a filing whose only
distinctly-larger text is the cover-page title). When that happens, the
"heading" swallows the ENTIRE rest of the document into one giant
raw_text blob with zero children - a single leaf node covering 50+ pages.
Since retrieval only ever sends the first ~3000 chars of a leaf node's
raw_text to the LLM, everything past the first page or two of that blob
becomes invisible to retrieval, even though it's sitting right there in
the DB.

To fix this, _expand_oversized_leaves() walks the finished tree and, for
any leaf node whose raw_text is larger than a threshold, replaces it with
page-based sub-chunk children (same shape as the "no headings at all"
fallback below) - so the tree-search can drill into the actual pages
that matter instead of only ever seeing the start of a giant blob.
"""
from collections import Counter

import pymupdf as fitz  # PyMuPDF (import name kept as `fitz` for readability below)

MAX_HEADING_LEVELS = 4
MAX_HEADING_WORDS = 20
BOLD_HEADING_MAX_WORDS = 12
PAGE_CHUNK_SIZE = 5

# If a node's own raw_text is longer than this, it gets split into
# page-based sub-chunks instead of being left as one giant leaf. Set
# somewhat above CONTEXT_CHARS_PER_NODE (3000, in retrieval_agent.py /
# multi_doc_service.py) so only genuinely oversized sections get split -
# a normal-sized section that already fits in one context window is left
# alone.
MAX_LEAF_TEXT_CHARS = 4000


def _is_bold_span(font_name: str, flags: int) -> bool:
    return bool(flags & (1 << 4)) or "bold" in font_name.lower()


def extract_lines(file_path: str) -> tuple[int, list[dict]]:
    doc = fitz.open(file_path)
    page_count = doc.page_count
    lines: list[dict] = []

    for page_index in range(page_count):
        page = doc[page_index]
        page_dict = page.get_text("dict")

        for block in page_dict.get("blocks", []):
            if block.get("type") != 0:
                continue
            for line in block.get("lines", []):
                text_parts, sizes, bold_flags = [], [], []
                for span in line.get("spans", []):
                    text = span.get("text", "")
                    if not text.strip():
                        continue
                    text_parts.append(text)
                    sizes.append(span.get("size", 0.0))
                    bold_flags.append(_is_bold_span(span.get("font", ""), span.get("flags", 0)))

                line_text = "".join(text_parts).strip()
                if not line_text:
                    continue

                lines.append({
                    "text": line_text,
                    "size": (sum(sizes) / len(sizes)) if sizes else 0.0,
                    "bold": all(bold_flags) if bold_flags else False,
                    "page": page_index + 1,
                })

    doc.close()
    return page_count, lines


def detect_headings(lines: list[dict]) -> list[dict]:
    if not lines:
        return []

    rounded_sizes = [round(l["size"]) for l in lines]
    body_size = Counter(rounded_sizes).most_common(1)[0][0]

    larger_sizes = sorted({s for s in rounded_sizes if s > body_size}, reverse=True)
    size_to_level = {size: i + 1 for i, size in enumerate(larger_sizes[:MAX_HEADING_LEVELS])}

    headings = []
    for idx, line in enumerate(lines):
        size = round(line["size"])
        word_count = len(line["text"].split())

        if size in size_to_level:
            if word_count > MAX_HEADING_WORDS:
                continue
            level = size_to_level[size]
        elif (
            not larger_sizes
            and line["bold"]
            and size >= body_size
            and word_count <= BOLD_HEADING_MAX_WORDS
            and not line["text"].endswith((".", ",", ";", ":"))
        ):
            level = 1
        else:
            continue

        headings.append({
            "line_index": idx,
            "level": level,
            "title": line["text"],
            "page": line["page"],
        })

    return headings


def _attach_direct_text(lines: list[dict], headings: list[dict]) -> None:
    for i, h in enumerate(headings):
        start = h["line_index"] + 1
        end = headings[i + 1]["line_index"] if i + 1 < len(headings) else len(lines)
        text_lines = [lines[j]["text"] for j in range(start, end)]
        h["raw_text"] = "\n".join(text_lines).strip()
        h["page_end"] = lines[end - 1]["page"] if end > start else h["page"]
        # Kept only for _expand_oversized_leaves() below - not part of the
        # public tree shape, stripped before returning from parse_pdf_to_tree.
        h["_own_lines"] = lines[start:end]


def _nest_headings(headings: list[dict]) -> list[dict]:
    roots: list[dict] = []
    stack: list[tuple[int, dict]] = []

    for h in headings:
        node = {
            "title": h["title"],
            "summary": None,
            "raw_text": h["raw_text"],
            "page_start": h["page"],
            "page_end": h["page_end"],
            "children": [],
            "_own_lines": h["_own_lines"],
        }
        while stack and stack[-1][0] >= h["level"]:
            stack.pop()
        if stack:
            stack[-1][1]["children"].append(node)
        else:
            roots.append(node)
        stack.append((h["level"], node))

    return roots


def _build_page_chunks(lines: list[dict], page_count: int) -> list[dict]:
    if not lines or page_count == 0:
        return []

    nodes = []
    for start_page in range(1, page_count + 1, PAGE_CHUNK_SIZE):
        end_page = min(start_page + PAGE_CHUNK_SIZE - 1, page_count)
        chunk_text = "\n".join(
            l["text"] for l in lines if start_page <= l["page"] <= end_page
        ).strip()
        title = f"Pages {start_page}-{end_page}" if end_page > start_page else f"Page {start_page}"
        nodes.append({
            "title": title,
            "summary": None,
            "raw_text": chunk_text,
            "page_start": start_page,
            "page_end": end_page,
            "children": [],
        })
    return nodes


def _build_page_chunks_from_lines(own_lines: list[dict], page_start: int, page_end: int) -> list[dict]:
    """
    Same idea as _build_page_chunks(), but scoped to one oversized node's
    own slice of lines/pages rather than the whole document. Used by
    _expand_oversized_leaves() to break up a giant leaf into page-sized
    children so retrieval can drill into it level by level.
    """
    if not own_lines or page_end < page_start:
        return []

    nodes = []
    for chunk_start in range(page_start, page_end + 1, PAGE_CHUNK_SIZE):
        chunk_end = min(chunk_start + PAGE_CHUNK_SIZE - 1, page_end)
        chunk_text = "\n".join(
            l["text"] for l in own_lines if chunk_start <= l["page"] <= chunk_end
        ).strip()
        if not chunk_text:
            continue
        title = f"Pages {chunk_start}-{chunk_end}" if chunk_end > chunk_start else f"Page {chunk_start}"
        nodes.append({
            "title": title,
            "summary": None,
            "raw_text": chunk_text,
            "page_start": chunk_start,
            "page_end": chunk_end,
            "children": [],
        })
    return nodes


def _expand_oversized_leaves(nodes: list[dict]) -> None:
    """
    Recursively walks a tree (in place) and, for any LEAF node (no
    children) whose raw_text is longer than MAX_LEAF_TEXT_CHARS, replaces
    its children with page-based sub-chunks built from that node's own
    lines. This is what prevents a single detected heading from silently
    swallowing an entire multi-page document into one untouchable blob -
    see the module docstring for why this matters.

    The parent node's own raw_text is left as-is (still usable as a
    fallback in retrieval_agent._walk() if none of the new children get
    selected), only its children list is populated.
    """
    for node in nodes:
        children = node.get("children") or []
        if children:
            _expand_oversized_leaves(children)
            continue

        raw_text = node.get("raw_text") or ""
        own_lines = node.pop("_own_lines", None)
        if len(raw_text) > MAX_LEAF_TEXT_CHARS and own_lines:
            sub_chunks = _build_page_chunks_from_lines(
                own_lines, node["page_start"], node["page_end"]
            )
            # Only worth it if this actually produces more than one chunk -
            # a single-chunk result would just be the same blob again.
            if len(sub_chunks) > 1:
                node["children"] = sub_chunks


def _strip_internal_keys(nodes: list[dict]) -> None:
    """Removes the temporary _own_lines key left on any node that never
    got expanded (e.g. it had children already, or wasn't oversized)."""
    for node in nodes:
        node.pop("_own_lines", None)
        if node.get("children"):
            _strip_internal_keys(node["children"])


def parse_pdf_to_tree(file_path: str) -> dict:
    page_count, lines = extract_lines(file_path)
    headings = detect_headings(lines)

    if headings:
        _attach_direct_text(lines, headings)
        tree = _nest_headings(headings)
        _expand_oversized_leaves(tree)
        _strip_internal_keys(tree)
    else:
        tree = _build_page_chunks(lines, page_count)

    return {"page_count": page_count, "tree": tree}

### backend/app/services/reranker_service.py
"""
Phase 6 (Chunk 4), Step 4a: lightweight embedding-based reranker.

The multi-document orchestrator (multi_doc_service.py) can end up with
more collected nodes than we want to stuff into the final answer prompt
(e.g. 18 nodes from 8 candidate documents). Previously these were simply
truncated in whatever order the candidate documents happened to be
processed in - not based on actual relevance.

This module replaces that naive cap with a real (but cheap) relevance
ranking: embed the question once, embed each candidate node's
title+summary (not the full raw_text - keeps embedding calls fast and
node summaries are already a concise stand-in for the node's content),
score by cosine similarity, and keep only the top N.

Deliberately has NO LLM /api/chat call in it - only /api/embed calls,
which are much cheaper and faster. This keeps reranking viable even when
there are dozens of candidate nodes.
"""
import asyncio
import logging
import math

from app.models.models import Document, TreeNode
from app.services.ollama_service import get_embedding

logger = logging.getLogger("pageindex.reranker")

DEFAULT_TOP_N = 8


def _cosine_similarity(a: list[float], b: list[float]) -> float:
    dot = sum(x * y for x, y in zip(a, b))
    norm_a = math.sqrt(sum(x * x for x in a))
    norm_b = math.sqrt(sum(y * y for y in b))
    if norm_a == 0 or norm_b == 0:
        return 0.0
    return dot / (norm_a * norm_b)


def _node_embedding_text(node: TreeNode) -> str:
    """
    What we actually embed for each node - title + summary, not the full
    raw_text. Keeps embedding calls fast and consistent in size regardless
    of how long a section's raw text is; the summary (from summary_service.py)
    is already a compact stand-in for what the section covers.
    """
    summary = node.summary or ""
    return f"{node.title}. {summary}".strip()


async def rerank_nodes(
    question: str,
    nodes_with_documents: list[tuple[Document, TreeNode]],
    top_n: int = DEFAULT_TOP_N,
) -> list[tuple[Document, TreeNode]]:
    """
    Scores each (document, node) pair by cosine similarity between the
    question's embedding and the node's title+summary embedding, and
    returns the top_n highest-scoring pairs, sorted best-first.

    If there are already <= top_n pairs, returns them unchanged (no need
    to spend embedding calls ranking something that isn't being cut).
    """
    if len(nodes_with_documents) <= top_n:
        logger.info(
            "Only %d nodes collected (<= top_n=%d) - skipping reranking, keeping all",
            len(nodes_with_documents), top_n,
        )
        return nodes_with_documents

    logger.info("Reranking %d nodes down to top %d via embeddings", len(nodes_with_documents), top_n)

    question_embedding = await get_embedding(question)

    node_embeddings = await asyncio.gather(*[
        get_embedding(_node_embedding_text(node))
        for _, node in nodes_with_documents
    ])

    scored = [
        (_cosine_similarity(question_embedding, node_emb), document, node)
        for (document, node), node_emb in zip(nodes_with_documents, node_embeddings)
    ]

    scored.sort(key=lambda triple: triple[0], reverse=True)

    top = scored[:top_n]

    logger.info(
        "Reranked top %d: %s",
        len(top),
        [(round(score, 3), doc.filename, node.title) for score, doc, node in top],
    )

    return [(document, node) for _, document, node in top]

### retrieval_agent.py

"""
Phase 4: tree-search retrieval agent.

Given a question, this walks a single document's PageIndex tree top-down:
  1. At the root level, ask the LLM which top-level sections (by title +
     summary) might contain the answer.
  2. For each selected section, recurse into its children and ask again -
     narrowing down level by level instead of scanning the whole document.
  3. Once a selected node has no children (or none of its children seem
     relevant), its raw text is collected as context.
  4. Once enough context is collected, ask the LLM to answer using ONLY
     that collected text.

This is deliberately plain recursive Python (no LangGraph) per the
project plan.

Two entry points are exported:
  - collect_relevant_nodes(): runs steps 1-3 only and returns the raw
    TreeNode list, with NO LLM synthesis. This is the primitive the
    multi-document orchestrators (multi_doc_service.py, graph.py) call
    once per candidate document, so they can merge nodes across many
    documents before doing ONE final synthesis over all of them.
  - answer_question(): the original single-document entry point (steps
    1-4) - collects nodes via collect_relevant_nodes() and then also
    synthesizes a final answer, for callers that just want one document
    answered on its own.
"""
import logging
import re

from sqlalchemy.orm import Session

from app.models.models import TreeNode
from app.services.ollama_service import chat_completion

logger = logging.getLogger("pageindex.retrieval")

MAX_SELECTIONS_PER_LEVEL = 3
MAX_LEAF_NODES = 5
MAX_DEPTH = 6
CONTEXT_CHARS_PER_NODE = 3000

SELECTION_SYSTEM_PROMPT = (
    "You are navigating a document's outline to find sections relevant to a "
    "question. You are shown a numbered list of section titles and short "
    "summaries. Reply with ONLY the numbers of sections that might help answer "
    "the question, separated by commas (example: 1,3). If none of them seem "
    "relevant, reply with exactly: NONE. Do not add any explanation."
)

ANSWER_SYSTEM_PROMPT = (
    "Answer the user's question using ONLY the document excerpts provided "
    "below. If the excerpts don't fully answer the question, say what you "
    "found and note what's missing. Do not use outside knowledge. Keep the "
    "answer concise and clear."
)

NO_CONTEXT_SYSTEM_PROMPT = (
    "Answer the question honestly. If you don't have enough information to "
    "answer it, say so plainly rather than guessing."
)


def _build_children_map(nodes: list[TreeNode]) -> dict[str | None, list[TreeNode]]:
    children_map: dict[str | None, list[TreeNode]] = {}
    for n in nodes:
        children_map.setdefault(n.parent_id, []).append(n)
    for key in children_map:
        children_map[key].sort(key=lambda n: n.order_index)
    return children_map


async def _select_relevant(question: str, candidates: list[TreeNode], level_label: str) -> list[TreeNode]:
    if not candidates:
        return []

    listing = "\n".join(
        f"{i + 1}. {n.title} - {n.summary or '(no summary yet)'}"
        for i, n in enumerate(candidates)
    )
    prompt = f"Question: {question}\n\nSections:\n{listing}\n\nWhich section numbers might contain the answer?"

    logger.info(
        "[%s] Asking LLM to pick among %d candidates: %s",
        level_label, len(candidates), [n.title for n in candidates],
    )

    response = await chat_completion(prompt=prompt, system=SELECTION_SYSTEM_PROMPT)

    numbers = sorted(set(int(x) for x in re.findall(r"\d+", response)))
    selected = [candidates[n - 1] for n in numbers if 1 <= n <= len(candidates)]
    selected = selected[:MAX_SELECTIONS_PER_LEVEL]

    logger.info(
        "[%s] LLM raw reply=%r -> parsed_numbers=%s -> selected=%s",
        level_label, response, numbers, [n.title for n in selected],
    )

    return selected


async def _walk(
    question: str,
    node: TreeNode,
    children_map: dict[str | None, list[TreeNode]],
    depth: int,
    collected: list[TreeNode],
) -> None:
    if len(collected) >= MAX_LEAF_NODES or depth > MAX_DEPTH:
        return

    children = children_map.get(node.id, [])
    logger.info("[depth=%d] Visiting node %r (%d children)", depth, node.title, len(children))

    if not children:
        logger.info("[depth=%d] Leaf node - collecting %r", depth, node.title)
        collected.append(node)
        return

    selected_children = await _select_relevant(question, children, level_label=f"depth={depth}")
    if not selected_children:
        if node.raw_text and node.raw_text.strip():
            logger.info("[depth=%d] No relevant children - falling back to own text of %r", depth, node.title)
            collected.append(node)
        else:
            logger.info("[depth=%d] No relevant children and no own text for %r - dropping branch", depth, node.title)
        return

    for child in selected_children:
        if len(collected) >= MAX_LEAF_NODES:
            break
        await _walk(question, child, children_map, depth + 1, collected)


async def collect_relevant_nodes(
    db: Session,
    document_id: str,
    question: str,
    nodes: list[TreeNode] | None = None,
) -> list[TreeNode]:
    """
    Runs the tree-walk (steps 1-3 above) against ONE document and returns
    the collected TreeNode rows relevant to `question` - no LLM synthesis,
    no DB writes. This is what multi_doc_service.py and graph.py call once
    per candidate document; they merge the results across every candidate
    before doing a single final synthesis over all of them.

    `nodes` lets a caller pass in an already-fetched list of this
    document's TreeNode rows (avoids a duplicate query when the caller
    already bulk-fetched them, e.g. multi_doc_service's
    _fetch_nodes_by_document). If omitted, they're queried here.
    """
    if nodes is None:
        nodes = db.query(TreeNode).filter(TreeNode.document_id == document_id).all()

    if not nodes:
        logger.info("Document %s has no tree nodes - nothing to collect", document_id)
        return []

    children_map = _build_children_map(nodes)
    roots = children_map.get(None, [])

    selected_roots = await _select_relevant(question, roots, level_label=f"doc={document_id} root")

    collected: list[TreeNode] = []
    for root in selected_roots:
        if len(collected) >= MAX_LEAF_NODES:
            break
        await _walk(question, root, children_map, depth=1, collected=collected)

    logger.info(
        "Document %s: collected %d node(s) for question %r: %s",
        document_id, len(collected), question, [n.title for n in collected],
    )
    return collected


async def answer_question(db: Session, document_id: str, question: str) -> dict:
    """
    Single-document entry point: collects relevant nodes for document_id
    via collect_relevant_nodes(), then synthesizes one answer from them.
    """
    logger.info("=== New question for document %s: %r ===", document_id, question)

    nodes = db.query(TreeNode).filter(TreeNode.document_id == document_id).all()

    if not nodes:
        logger.info("No tree nodes found for document %s - falling back to plain answer", document_id)
        answer = await chat_completion(
            prompt=question,
            system="No document content is available yet for this document.",
        )
        return {"answer": answer, "cited_node_ids": []}

    collected = await collect_relevant_nodes(db, document_id, question, nodes=nodes)

    logger.info("Final collected context nodes: %s", [n.title for n in collected])

    if not collected:
        answer = await chat_completion(
            prompt=f"Question: {question}\n\n(No relevant section of the document was found.)",
            system=NO_CONTEXT_SYSTEM_PROMPT,
        )
        return {"answer": answer, "cited_node_ids": []}

    context_blocks = []
    for n in collected:
        text = (n.raw_text or "").strip() or "(no direct text - see subsections)"
        context_blocks.append(f"### {n.title} (pages {n.page_start}-{n.page_end})\n{text[:CONTEXT_CHARS_PER_NODE]}")
    context = "\n\n".join(context_blocks)

    final_prompt = f"Document excerpts:\n\n{context}\n\nQuestion: {question}"
    answer = await chat_completion(prompt=final_prompt, system=ANSWER_SYSTEM_PROMPT)

    return {"answer": answer, "cited_node_ids": [n.id for n in collected]}

"""
Phase 3 (extended in Phase 6): generates a short summary AND keywords for
every TreeNode of a document, then rolls those up into one document-level
summary+keywords (used later for corpus-wide candidate narrowing).

Runs bottom-up (children before parents) so a parent node with no direct
text of its own gets summarized from its children's summaries instead of
being left blank.

import asyncio
import json
import logging
import re

from sqlalchemy.orm import Session

from app.models.models import TreeNode, Document
from app.services.ollama_service import chat_completion

logger = logging.getLogger("pageindex.summary")

SUMMARY_SYSTEM_PROMPT = (
    "You summarize one section of a document for a retrieval system. "
    "Given the section title and its text, reply in EXACTLY this format, nothing else:\n"
    "SUMMARY: <one short sentence, max 25 words, describing what this section covers>\n"
    "KEYWORDS: <5-8 comma-separated keywords or key phrases capturing the important "
    "terms, names, and numbers mentioned, so nothing important is missed>"
)

DOC_ROLLUP_SYSTEM_PROMPT = (
    "You write a short overview of an entire document for a search index, based on "
    "summaries of its top-level sections. Reply in EXACTLY this format, nothing else:\n"
    "SUMMARY: <2-3 sentences describing what this document is about overall>\n"
    "KEYWORDS: <8-12 comma-separated keywords covering the document's main topics, "
    "entities, and terms>"
)

MAX_CONTENT_CHARS = 4000
MAX_ROOTS_FOR_ROLLUP = 20


def _parse_summary_response(response: str) -> tuple[str, list[str]]:
    """
    Robust line-based parser for the SUMMARY:/KEYWORDS: format above.
    Falls back to treating the whole response as the summary if the model
    doesn't follow the format at all.
    """
    summary = ""
    keywords: list[str] = []
    for line in response.splitlines():
        line = line.strip()
        if not line:
            continue
        lower = line.lower()
        if lower.startswith("summary"):
            _, _, rest = line.partition(":")
            summary = rest.strip()
        elif lower.startswith("keyword"):
            _, _, rest = line.partition(":")
            raw = rest.strip()
            parts = re.split(r"[;,]", raw)
            keywords = [p.strip() for p in parts if p.strip()]
    if not summary:
        summary = response.strip()
    return summary, keywords


def _build_children_map(nodes: list[TreeNode]) -> dict[str | None, list[TreeNode]]:
    children_map: dict[str | None, list[TreeNode]] = {}
    for n in nodes:
        children_map.setdefault(n.parent_id, []).append(n)
    for key in children_map:
        children_map[key].sort(key=lambda n: n.order_index)
    return children_map


async def _summarize_node(node: TreeNode, children_summaries: list[str]) -> dict:
    if node.raw_text and node.raw_text.strip():
        content = node.raw_text[:MAX_CONTENT_CHARS]
    elif children_summaries:
        content = "\n".join(f"- {s}" for s in children_summaries)
    else:
        content = "(no content)"

    prompt = f"Section title: {node.title}\n\nContent:\n{content}"
    response = await chat_completion(prompt=prompt, system=SUMMARY_SYSTEM_PROMPT)
    summary, keywords = _parse_summary_response(response)
    return {"summary": summary or node.title, "keywords": keywords}


async def _summarize_tree_async(db: Session, document_id: str) -> None:
    nodes = db.query(TreeNode).filter(TreeNode.document_id == document_id).all()
    if not nodes:
        return

    children_map = _build_children_map(nodes)
    by_id = {n.id: n for n in nodes}
    summaries: dict[str, str] = {}

    def post_order(parent_id: str | None) -> list[str]:
        order = []
        for child in children_map.get(parent_id, []):
            order.extend(post_order(child.id))
            order.append(child.id)
        return order

    ordered_ids = post_order(None)

    for node_id in ordered_ids:
        node = by_id[node_id]
        child_summaries = [summaries[c.id] for c in children_map.get(node_id, [])]
        result = await _summarize_node(node, child_summaries)
        summaries[node_id] = result["summary"]
        node.summary = result["summary"]
        node.keywords = json.dumps(result["keywords"])
        db.add(node)

    db.commit()


async def _generate_document_rollup(db: Session, document_id: str) -> None:
    roots = (
        db.query(TreeNode)
        .filter(TreeNode.document_id == document_id, TreeNode.parent_id.is_(None))
        .order_by(TreeNode.order_index)
        .limit(MAX_ROOTS_FOR_ROLLUP)
        .all()
    )
    if not roots:
        return

    listing = "\n".join(f"- {n.title}: {n.summary or '(no summary)'}" for n in roots)
    prompt = f"Section summaries:\n{listing}"

    response = await chat_completion(prompt=prompt, system=DOC_ROLLUP_SYSTEM_PROMPT)
    summary, keywords = _parse_summary_response(response)

    document = db.query(Document).filter(Document.id == document_id).first()
    if document:
        document.summary = summary or document.filename
        document.keywords = json.dumps(keywords)
        db.add(document)
        db.commit()
        logger.info("Document %s rollup: summary=%r keywords=%s", document_id, summary, keywords)


def generate_summaries_for_document(db: Session, document_id: str) -> None:
    """Sync wrapper - safe to call from a sync context (e.g. a background task).
    Generates per-node summaries+keywords, then rolls them up into one
    document-level summary+keywords."""
    async def _run():
        await _summarize_tree_async(db, document_id)
        await _generate_document_rollup(db, document_id)

    asyncio.run(_run())

import React, { useState } from "react";

function TreeNode({ node, depth }) {
  const [expanded, setExpanded] = useState(depth < 1);
  const hasChildren = node.children && node.children.length > 0;

  return (
    <div style={{ marginLeft: depth > 0 ? 14 : 0 }}>
      <div
        className="tree-node-row"
        onClick={() => hasChildren && setExpanded((e) => !e)}
        style={{ cursor: hasChildren ? "pointer" : "default" }}
      >
        {hasChildren && (
          <span className="tree-toggle">{expanded ? "▾" : "▸"}</span>
        )}
        {!hasChildren && (
          <span className="tree-toggle tree-toggle-leaf">•</span>
        )}
        <span className="tree-title">{node.title}</span>
        {(node.page_start || node.page_end) && (
          <span className="tree-pages">
            p{node.page_start}
            {node.page_end && node.page_end !== node.page_start
              ? `-${node.page_end}`
              : ""}
          </span>
        )}
      </div>
      {node.summary && <div className="tree-summary">{node.summary}</div>}
      {hasChildren && expanded && (
        <div>
          {node.children.map((child) => (
            <TreeNode key={child.id} node={child} depth={depth + 1} />
          ))}
        </div>
      )}
    </div>
  );
}

    return (
      <p style={{ color: "var(--text-dim)", fontSize: 13 }}>
        No structure available yet.
      </p>
    );
  }

  return (
    <div className="tree-view">
      {tree.map((node) => (
        <TreeNode key={node.id} node={node} depth={0} />
      ))}
    </div>
  );
