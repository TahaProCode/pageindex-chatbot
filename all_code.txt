"""
backend/app/graph.py

LangGraph-powered Autonomous Multi-Agent Framework for PageIndex RAG.

Key Capabilities:
1. Decomposes complex/cross-document queries into precise sub-queries.
2. Parallel context retrieval using Tree-Walking & Corpus Narrowing,
   scoped to a single upload-batch (batch_id).
3. Reranks & evaluates context completeness dynamically.
4. Auto-correcting feedback loop for missing context (Max Hops control).
5. Synthesizes 99%+ accurate answers with complete source attribution/citations.
"""

import asyncio
import logging
import json
import re
from typing import List, Dict, Any, Optional, TypedDict

from sqlalchemy.orm import Session
from langgraph.graph import StateGraph, END

# Import existing core services from your codebase
from app.models.models import Document, TreeNode
from app.services.corpus_service import narrow_candidate_documents
from app.services.retrieval_agent import collect_relevant_nodes
from app.services.reranker_service import rerank_nodes
from app.services.ollama_service import chat_completion

logger = logging.getLogger("pageindex.langgraph")

# Configuration Constants
MAX_TOTAL_CONTEXT_NODES = 10
CONTEXT_CHARS_PER_NODE = 3000
MAX_HOPS = 3

# Prompts
DECOMPOSE_SYSTEM_PROMPT = (
    "You are an expert query analyzer. Your job is to break down a user query into concise, independent "
    "sub-queries required to search across a document database. If the query asks for comparisons or multi-part "
    "information, break it into individual sub-questions. If the query is simple, reply with just the original query.\n"
    "Respond ONLY with a JSON array of string queries. Example:\n"
    '["What is Company X revenue for Q3?", "What is Company Y revenue for Q3?"]'
)

EVALUATE_SYSTEM_PROMPT = (
    "You are an objective auditor evaluating retrieved document excerpts against a user query.\n"
    "Reply in EXACTLY ONE of these formats:\n"
    "COMPLETE\n"
    "or\n"
    "INCOMPLETE: <short description of missing information to search next>\n\n"
    "Only mark INCOMPLETE if crucial facts requested by the user are missing."
)

SYNTHESIS_SYSTEM_PROMPT = (
    "You are an elite research analyst. Synthesize a complete, structured, and 100% factual answer to the "
    "user query using ONLY the provided document excerpts. Explicitly attribute facts to their source document filename.\n"
    "If the query involves comparisons or tables, format the answer cleanly using Markdown tables.\n"
    "Do not extrapolate or invent facts beyond the excerpts."
)


# ==========================================
# 1. GRAPH STATE DEFINITION
# ==========================================
class RAGAgentState(TypedDict):
    db: Session
    batch_id: str                       # required - which upload-batch this chat is scoped to
    document_id: Optional[str]          # populated only for legacy single-document mode
    original_query: str
    sub_queries: List[str]
    accumulated_context: List[Dict[str, Any]]  # Stores {"document": Doc, "node": Node}
    reranked_context: List[Dict[str, Any]]
    missing_info_query: Optional[str]
    hop_count: int
    is_complete: bool
    final_answer: str
    sources: List[Dict[str, Any]]


# ==========================================
# 2. HELPER FUNCTIONS
# ==========================================
def _fetch_nodes_by_document(db: Session, document_ids: List[str]) -> Dict[str, List[TreeNode]]:
    if not document_ids:
        return {}
    all_nodes = db.query(TreeNode).filter(TreeNode.document_id.in_(document_ids)).all()
    by_doc: Dict[str, List[TreeNode]] = {doc_id: [] for doc_id in document_ids}
    for node in all_nodes:
        by_doc.setdefault(node.document_id, []).append(node)
    return by_doc


async def _collect_for_doc(db: Session, doc: Document, query: str, nodes: List[TreeNode]):
    if not nodes:
        return doc, []
    collected = await collect_relevant_nodes(db, doc.id, query, nodes=nodes)
    return doc, collected


def _build_context_text(nodes_with_docs: List[Dict[str, Any]]) -> tuple[str, List[Dict[str, Any]]]:
    context_blocks = []
    sources = []
    for item in nodes_with_docs:
        doc: Document = item["document"]
        node: TreeNode = item["node"]
        text = (node.raw_text or "").strip() or "(no direct text)"
        context_blocks.append(
            f"### [{doc.filename}] {node.title} (pages {node.page_start}-{node.page_end})\n"
            f"{text[:CONTEXT_CHARS_PER_NODE]}"
        )
        sources.append({
            "document_id": doc.id,
            "filename": doc.filename,
            "node_id": node.id,
            "title": node.title,
            "page_start": node.page_start,
            "page_end": node.page_end,
        })
    return "\n\n".join(context_blocks), sources


# ==========================================
# 3. GRAPH NODE FUNCTIONS
# ==========================================
async def decompose_query_node(state: RAGAgentState) -> Dict[str, Any]:
    """Node 1: Breaks complex query into sub-queries."""
    query = state["original_query"]
    logger.info("--- [LangGraph] Decomposing Query (batch=%s): %r ---", state.get("batch_id"), query)

    # If scoped to a single document, skip heavy multi-query decomposition
    if state.get("document_id"):
        return {"sub_queries": [query], "hop_count": 0, "accumulated_context": []}

    prompt = f"User Query: {query}"
    response = await chat_completion(prompt=prompt, system=DECOMPOSE_SYSTEM_PROMPT)

    try:
        match = re.search(r"\[.*\]", response, re.DOTALL)
        if match:
            sub_queries = json.loads(match.group(0))
        else:
            sub_queries = [query]
    except Exception as e:
        logger.warning("Failed to parse sub-queries JSON: %s. Using original query.", e)
        sub_queries = [query]

    logger.info("Generated Sub-Queries: %s", sub_queries)
    return {"sub_queries": sub_queries, "hop_count": 0, "accumulated_context": []}


async def parallel_retrieval_node(state: RAGAgentState) -> Dict[str, Any]:
    """Node 2: Executes parallel tree retrieval across all active sub-queries."""
    db = state["db"]
    doc_id = state.get("document_id")
    batch_id = state["batch_id"]
    sub_queries = state.get("sub_queries", [state["original_query"]])
    current_hop = state.get("hop_count", 0) + 1

    logger.info("--- [LangGraph] Retrieval Pass Hop %d/%d (batch=%s) ---", current_hop, MAX_HOPS, batch_id)

    new_gathered: List[Dict[str, Any]] = []

    if doc_id:
        # Legacy Single Document Mode
        doc = db.query(Document).filter(Document.id == doc_id).first()
        nodes = db.query(TreeNode).filter(TreeNode.document_id == doc_id).all()
        for sq in sub_queries:
            _, collected = await _collect_for_doc(db, doc, sq, nodes)
            for n in collected:
                new_gathered.append({"document": doc, "node": n})
    else:
        # Batch-Scoped Multi-Document Mode - only documents belonging to
        # batch_id are ever considered as candidates.
        for sq in sub_queries:
            candidates = await narrow_candidate_documents(db, sq, batch_id=batch_id)
            if not candidates:
                continue
            cand_ids = [c.id for c in candidates]
            nodes_by_doc = _fetch_nodes_by_document(db, cand_ids)

            tasks = [_collect_for_doc(db, c, sq, nodes_by_doc.get(c.id, [])) for c in candidates]
            results = await asyncio.gather(*tasks)

            for d, collected_nodes in results:
                for n in collected_nodes:
                    new_gathered.append({"document": d, "node": n})

    # Deduplicate accumulated context by node.id
    existing_items = state.get("accumulated_context", [])
    seen_node_ids = {item["node"].id for item in existing_items}

    merged = list(existing_items)
    for item in new_gathered:
        if item["node"].id not in seen_node_ids:
            seen_node_ids.add(item["node"].id)
            merged.append(item)

    logger.info("Total accumulated context nodes after Hop %d: %d", current_hop, len(merged))
    return {"accumulated_context": merged, "hop_count": current_hop}


async def evaluate_and_rerank_node(state: RAGAgentState) -> Dict[str, Any]:
    """Node 3: Reranks gathered context using embeddings & audits completeness."""
    accumulated = state.get("accumulated_context", [])
    query = state["original_query"]
    hop_count = state.get("hop_count", 1)

    if not accumulated:
        logger.warning("No context gathered yet.")
        return {"is_complete": False, "missing_info_query": query, "reranked_context": []}

    # Transform for reranker service format: List[tuple[Document, TreeNode]]
    raw_pairs = [(item["document"], item["node"]) for item in accumulated]
    reranked_pairs = await rerank_nodes(query, raw_pairs, top_n=MAX_TOTAL_CONTEXT_NODES)

    reranked_items = [{"document": d, "node": n} for d, n in reranked_pairs]
    context_text, _ = _build_context_text(reranked_items)

    # Stop condition check if max hops reached
    if hop_count >= MAX_HOPS:
        logger.info("Max hops (%d) reached. Proceeding to answer.", MAX_HOPS)
        return {"is_complete": True, "reranked_context": reranked_items, "missing_info_query": None}

    # Completeness audit
    eval_prompt = f"User Question: {query}\n\nGathered Excerpts:\n{context_text}\n\nIs this sufficient?"
    response = await chat_completion(prompt=eval_prompt, system=EVALUATE_SYSTEM_PROMPT)

    stripped = response.strip()
    match = re.match(r"INCOMPLETE\s*:\s*(.+)", stripped, re.IGNORECASE | re.DOTALL)

    if match:
        missing_query = match.group(1).strip()
        logger.info("Evaluation: INCOMPLETE. Gap identified: %r", missing_query)
        return {
            "is_complete": False,
            "missing_info_query": missing_query,
            "sub_queries": [missing_query],  # Overwrite sub-queries for next hop loop
            "reranked_context": reranked_items,
        }

    logger.info("Evaluation: COMPLETE context verified.")
    return {"is_complete": True, "missing_info_query": None, "reranked_context": reranked_items}


async def synthesize_answer_node(state: RAGAgentState) -> Dict[str, Any]:
    """Node 4: Synthesizes final response with verified citations."""
    reranked = state.get("reranked_context", [])
    query = state["original_query"]

    if not reranked:
        return {
            "final_answer": "I could not find relevant information in the uploaded documents to answer your query.",
            "sources": [],
        }

    context_text, sources = _build_context_text(reranked)
    final_prompt = f"Document Excerpts:\n\n{context_text}\n\nUser Question: {query}"

    logger.info("Synthesizing final response from %d context nodes...", len(reranked))
    final_answer = await chat_completion(prompt=final_prompt, system=SYNTHESIS_SYSTEM_PROMPT)

    return {"final_answer": final_answer, "sources": sources}


# ==========================================
# 4. CONDITIONAL ROUTER
# ==========================================
def completion_router(state: RAGAgentState) -> str:
    """Routes execution: If complete or max hops hit -> Synthesize, else Loop Back."""
    if state.get("is_complete") or state.get("hop_count", 0) >= MAX_HOPS:
        return "synthesize"
    return "retrieve"


# ==========================================
# 5. GRAPH BUILDER & EXECUTOR
# ==========================================
def build_rag_graph() -> StateGraph:
    workflow = StateGraph(RAGAgentState)

    # Add Nodes
    workflow.add_node("decompose", decompose_query_node)
    workflow.add_node("retrieve", parallel_retrieval_node)
    workflow.add_node("evaluate", evaluate_and_rerank_node)
    workflow.add_node("synthesize", synthesize_answer_node)

    # Set Entry Point
    workflow.set_entry_point("decompose")

    # Connect Edges
    workflow.add_edge("decompose", "retrieve")
    workflow.add_edge("retrieve", "evaluate")

    # Conditional Loop Edge
    workflow.add_conditional_edges(
        "evaluate",
        completion_router,
        {
            "synthesize": "synthesize",
            "retrieve": "retrieve",
        },
    )

    workflow.add_edge("synthesize", END)
    return workflow.compile()


# Global compiled instance
langgraph_app = build_rag_graph()


# Interface function to trigger graph execution
async def run_langgraph_chat(
    db: Session,
    query: str,
    batch_id: str,
    document_id: Optional[str] = None,
) -> Dict[str, Any]:
    """
    Main entry point for API routers to execute LangGraph agent execution.

    batch_id is now required: every chat is scoped to one upload-batch, and
    cross-document retrieval only ever searches within that batch's
    documents. document_id is kept as an optional override for legacy
    single-document scoping and, when provided, must belong to batch_id.
    """
    initial_state: RAGAgentState = {
        "db": db,
        "batch_id": batch_id,
        "document_id": document_id,
        "original_query": query,
        "sub_queries": [],
        "accumulated_context": [],
        "reranked_context": [],
        "missing_info_query": None,
        "hop_count": 0,
        "is_complete": False,
        "final_answer": "",
        "sources": [],
    }

    result_state = await langgraph_app.ainvoke(initial_state)

    return {
        "answer": result_state.get("final_answer"),
        "sources": result_state.get("sources", []),
        "hops_used": result_state.get("hop_count", 1),
    }

"""
ORM models for the PageIndex chatbot.

Batch       -> one upload-batch (a group of documents uploaded together).
               A ChatSession is scoped to a Batch, so a single query can
               retrieve context across every document in that batch.
Document    -> one uploaded PDF, belongs to exactly one Batch
TreeNode    -> one node of the hierarchical PageIndex tree for a document
               (self-referencing parent_id builds the tree)
ChatSession -> one chat thread tied to a Batch
Message     -> one message (user or assistant) within a chat session
"""
import enum
import uuid
from datetime import datetime

from sqlalchemy import (
    Column, String, Integer, Text, DateTime, ForeignKey, Enum
)
from sqlalchemy.dialects.postgresql import UUID
from sqlalchemy.orm import relationship

from app.db.database import Base


def gen_uuid():
    return str(uuid.uuid4())


class DocumentStatus(str, enum.Enum):
    uploaded = "uploaded"
    parsing = "parsing"
    indexing = "indexing"
    ready = "ready"
    failed = "failed"


class Batch(Base):
    __tablename__ = "batches"

    id = Column(UUID(as_uuid=False), primary_key=True, default=gen_uuid)
    name = Column(String, nullable=True)
    created_at = Column(DateTime, default=datetime.utcnow)

    documents = relationship("Document", back_populates="batch", cascade="all, delete-orphan")
    chat_sessions = relationship("ChatSession", back_populates="batch", cascade="all, delete-orphan")


class Document(Base):
    __tablename__ = "documents"

    id = Column(UUID(as_uuid=False), primary_key=True, default=gen_uuid)
    batch_id = Column(UUID(as_uuid=False), ForeignKey("batches.id"), nullable=False)
    filename = Column(String, nullable=False)
    file_path = Column(String, nullable=False)
    status = Column(Enum(DocumentStatus), default=DocumentStatus.uploaded, nullable=False)
    page_count = Column(Integer, nullable=True)
    error_message = Column(Text, nullable=True)
    summary = Column(Text, nullable=True)     # document-level rollup summary (Phase 6)
    keywords = Column(Text, nullable=True)    # JSON-encoded list (Phase 6)
    created_at = Column(DateTime, default=datetime.utcnow)

    batch = relationship("Batch", back_populates="documents")
    nodes = relationship("TreeNode", back_populates="document", cascade="all, delete-orphan")


class TreeNode(Base):
    __tablename__ = "tree_nodes"

    id = Column(UUID(as_uuid=False), primary_key=True, default=gen_uuid)
    document_id = Column(UUID(as_uuid=False), ForeignKey("documents.id"), nullable=False)
    parent_id = Column(UUID(as_uuid=False), ForeignKey("tree_nodes.id"), nullable=True)

    title = Column(String, nullable=False)
    summary = Column(Text, nullable=True)       # LLM-generated short summary
    keywords = Column(Text, nullable=True)      # JSON-encoded list (Phase 6)
    raw_text = Column(Text, nullable=True)       # full text belonging to this node
    page_start = Column(Integer, nullable=True)
    page_end = Column(Integer, nullable=True)
    order_index = Column(Integer, default=0)     # sibling ordering

    document = relationship("Document", back_populates="nodes")
    children = relationship(
        "TreeNode",
        backref="parent",
        remote_side=[id],
        cascade="all, delete-orphan",
        single_parent=True,
    )


class ChatSession(Base):
    __tablename__ = "chat_sessions"

    id = Column(UUID(as_uuid=False), primary_key=True, default=gen_uuid)
    batch_id = Column(UUID(as_uuid=False), ForeignKey("batches.id"), nullable=False)
    created_at = Column(DateTime, default=datetime.utcnow)

    batch = relationship("Batch", back_populates="chat_sessions")
    messages = relationship("Message", back_populates="session", cascade="all, delete-orphan")


class MessageRole(str, enum.Enum):
    user = "user"
    assistant = "assistant"


class Message(Base):
    __tablename__ = "messages"

    id = Column(UUID(as_uuid=False), primary_key=True, default=gen_uuid)

    db: Session = Depends(get_db),

    Accepts one or more PDFs in a single request, creates one Batch row,
    and creates a Document row per file under that batch. Every document

    batch = Batch(name=batch_name)

    db.commit()
    db.refresh(batch)

        stored_name = f"{uuid.uuid4()}_{f.filename}"
        dest_path = os.path.join(settings.upload_dir, stored_name)

            file_path=dest_path,
            status=DocumentStatus.uploaded,

"""
Corpus-wide candidate narrowing, scoped to a single upload batch.

When a user asks a question we cannot afford full tree-search on every
document. This module does a cheap pass first using document-level
summary + keywords, then (if the LLM finds nothing) falls back to a
simple BM25-style keyword ranking so we never silently return "most
recent documents" that have nothing to do with the query.
"""
import asyncio
import json
import logging
import math
import re
from collections import Counter

from sqlalchemy.orm import Session

from app.core.config import settings
from app.models.models import Document, DocumentStatus
from app.services.ollama_service import chat_completion

logger = logging.getLogger("pageindex.corpus")

MAX_CANDIDATE_DOCUMENTS = 15
DOCUMENTS_PER_BATCH = 40

CANDIDATE_SELECTION_SYSTEM_PROMPT = (
    "You are choosing which documents are worth searching to answer a "
    "question. You are shown a numbered list of documents, each with a "
    "short summary and keywords. Reply with ONLY the numbers of documents "
    "that might contain relevant information, separated by commas "
    "(example: 1,3,4). Include a document even if you're only somewhat "
    "sure it's relevant - it's fine to include more than the minimum. "
    "If truly none of them seem relevant, reply with exactly: NONE. "
    "Do not add any explanation."
)


def _parse_keywords(raw: str | None) -> list[str]:
    if not raw:
        return []
    try:
        return json.loads(raw)
    except (TypeError, ValueError):
        return []


def _format_document_listing(documents: list[Document]) -> str:
    lines = []
    for i, doc in enumerate(documents):
        keywords = _parse_keywords(doc.keywords)
        keywords_str = ", ".join(keywords) if keywords else "(none)"
        summary = doc.summary or "(no summary available)"
        lines.append(
            f"{i + 1}. {doc.filename}\n   Summary: {summary}\n   Keywords: {keywords_str}"
        )
    return "\n".join(lines)


def _chunk_list(items: list, size: int) -> list[list]:
    return [items[i : i + size] for i in range(0, len(items), size)]


def _tokenize(text: str) -> list[str]:
    return re.findall(r"[a-z0-9]{2,}", (text or "").lower())


def _bm25_rank(
    question: str,
    documents: list[Document],
    top_n: int,
) -> list[Document]:
    """
    Lightweight BM25-inspired ranking over document filename + summary + keywords.
    Used as fallback when the LLM narrowing returns nothing relevant.
    """
    query_tokens = _tokenize(question)
    if not query_tokens or not documents:
        return documents[:top_n]

    # Document "text" for scoring
    doc_texts = []
    for doc in documents:
        parts = [doc.filename or ""]
        if doc.summary:
            parts.append(doc.summary)
        kws = _parse_keywords(doc.keywords)
        parts.extend(kws)
        doc_texts.append(" ".join(parts))

    tokenized_docs = [_tokenize(t) for t in doc_texts]
    doc_lens = [len(td) or 1 for td in tokenized_docs]
    avgdl = sum(doc_lens) / len(doc_lens)

    # Document frequency
    df: Counter = Counter()
    for td in tokenized_docs:
        df.update(set(td))

    N = len(documents)
    k1 = 1.5
    b = 0.75

    scores = []
    for i, td in enumerate(tokenized_docs):
        tf = Counter(td)
        score = 0.0
        for qt in query_tokens:
            if qt not in tf:
                continue
            idf = math.log(1 + (N - df[qt] + 0.5) / (df[qt] + 0.5))
            freq = tf[qt]
            denom = freq + k1 * (1 - b + b * doc_lens[i] / avgdl)
            score += idf * (freq * (k1 + 1)) / denom
        scores.append((score, documents[i]))

    scores.sort(key=lambda x: x[0], reverse=True)
    ranked = [doc for sc, doc in scores if sc > 0][:top_n]

    logger.info(
        "BM25 fallback ranked %d docs (top scores): %s",
        len(ranked),
        [(round(sc, 3), d.filename) for sc, d in scores[:5]],
    )
    # If BM25 found nothing, fall back to most recent as last resort
    if not ranked:
        return documents[:top_n]
    return ranked


async def _narrow_one_batch(
    question: str, batch: list[Document], batch_label: str
) -> list[Document]:
    listing = _format_document_listing(batch)
    prompt = (
        f"Question: {question}\n\nDocuments:\n{listing}\n\n"
        "Which document numbers might contain the answer?"
    )

    logger.info(
        "[%s] Asking LLM to narrow among %d documents: %s",
        batch_label, len(batch), [d.filename for d in batch],
    )

    response = await chat_completion(prompt=prompt, system=CANDIDATE_SELECTION_SYSTEM_PROMPT)

    numbers = sorted(set(int(x) for x in re.findall(r"\b\d+\b", response)))
    selected = [batch[n - 1] for n in numbers if 1 <= n <= len(batch)]

    logger.info(
        "[%s] LLM raw reply=%r -> parsed_numbers=%s -> picked=%s",
        batch_label, response, numbers, [d.filename for d in selected],
    )
    return selected


async def narrow_candidate_documents(
    db: Session,
    question: str,
    batch_id: str,
    max_candidates: int | None = None,
    documents_per_batch: int = DOCUMENTS_PER_BATCH,
) -> list[Document]:
    """
    Returns a shortlist of Document rows worth running the full tree-search
    on for this question, scoped to a single upload-batch.
    """
    max_candidates = max_candidates or settings.max_candidate_documents

    documents = (
        db.query(Document)
        .filter(Document.batch_id == batch_id, Document.status == DocumentStatus.ready)
        .order_by(Document.created_at.desc())
        .all()
    )

    if not documents:
        logger.info("No ready documents in batch %s - nothing to narrow", batch_id)
        return []

    if len(documents) <= max_candidates:
        logger.info(
            "Batch %s has only %d ready documents (<= max_candidates=%d) - "
            "skipping LLM narrowing, using all of them",
            batch_id, len(documents), max_candidates,
        )
        return documents

    chunks = _chunk_list(documents, documents_per_batch)
    logger.info(
        "Narrowing %d documents in batch %s across %d chunk(s) of up to %d each (parallel)",
        len(documents), batch_id, len(chunks), documents_per_batch,
    )

    chunk_results = await asyncio.gather(*[
        _narrow_one_batch(
            question, chunk,
            batch_label=f"batch={batch_id} chunk={i + 1}/{len(chunks)}",
        )
        for i, chunk in enumerate(chunks)
    ])

    seen_ids: set[str] = set()
    merged: list[Document] = []
    for chunk_selected in chunk_results:
        for doc in chunk_selected:
            if doc.id not in seen_ids:
                seen_ids.add(doc.id)
                merged.append(doc)

    logger.info(
        "Merged candidates across all chunks for batch %s: %d documents -> %s",
        batch_id, len(merged), [d.filename for d in merged],
    )

    if not merged:
        # BM25 / keyword ranking fallback instead of blind "most recent"
        logger.info(
            "LLM found no relevant documents - applying BM25 keyword ranking fallback"
        )
        return _bm25_rank(question, documents, max_candidates)

    if len(merged) > max_candidates:
        logger.info(
            "Merged candidates (%d) exceed max_candidates (%d) - truncating",
            len(merged), max_candidates,
        )
        merged = merged[:max_candidates]

    return merged

"""
Multi-worker indexing queue.

Documents are processed with controlled concurrency (default 3 workers)
so a 200-file batch finishes in a fraction of the time of a single-threaded
queue, while still protecting the local Ollama instance from being
saturated by dozens of concurrent indexing jobs.
"""
import logging
import queue
import threading

from app.core.config import settings
from app.services.indexing_service import process_document

logger = logging.getLogger("pageindex.queue")

_queue: "queue.Queue[str]" = queue.Queue()
_workers_started = False
_lock = threading.Lock()

# Safe default if config.py was not updated yet
DEFAULT_MAX_CONCURRENT_INDEXING = 3


def _worker_loop(worker_id: int) -> None:
    while True:
        document_id = _queue.get()
        try:
            logger.info(
                "Indexing worker-%d: starting document %s (%d remaining in queue)",
                worker_id, document_id, _queue.qsize(),
            )
            process_document(document_id)
            logger.info(
                "Indexing worker-%d: finished document %s",
                worker_id, document_id,
            )
        except Exception:
            logger.exception(
                "Indexing worker-%d: unhandled error processing document %s",
                worker_id, document_id,
            )
        finally:
            _queue.task_done()


def start_worker() -> None:
    """
    Call once at app startup. Spawns concurrent worker threads.
    Safe to call multiple times (idempotent).
    """
    global _workers_started
    with _lock:
        if _workers_started:
            return

            )
            t.start()

            n_workers,

def _fetch_nodes_by_document(db: Session, document_ids: list[str]) -> dict[str, list[TreeNode]]:
    if not document_ids:
        return {}
    all_nodes = (
        db.query(TreeNode)
        .filter(TreeNode.document_id.in_(document_ids))
        .all()
    )
    by_document: dict[str, list[TreeNode]] = {doc_id: [] for doc_id in document_ids}
    for node in all_nodes:
        by_document.setdefault(node.document_id, []).append(node)
    return by_document


async def _collect_for_document(

    search_query: str,
    nodes: list[TreeNode],

async def _gather_context(
    db: Session,
    search_query: str,
    batch_id: str,
    hop_label: str,
) -> list[tuple[Document, TreeNode]]:
    candidates = await narrow_candidate_documents(db, search_query, batch_id=batch_id)
    if not candidates:
        logger.info(
            "[%s] No candidate documents found in batch %s for query %r",
            hop_label, batch_id, search_query,
        )
        return []

    candidate_ids = [doc.id for doc in candidates]
    nodes_by_document = _fetch_nodes_by_document(db, candidate_ids)

    logger.info(
        "[%s] Running tree-search in parallel across %d candidate documents "
        "(batch=%s) for query %r: %s",
        hop_label, len(candidates), batch_id, search_query,
        [d.filename for d in candidates],
    )

    results = await asyncio.gather(*[
        _collect_for_document(db, doc, search_query, nodes_by_document.get(doc.id, []))
        for doc in candidates
    ])

    gathered: list[tuple[Document, TreeNode]] = []
    for document, collected_nodes in results:
        for node in collected_nodes:
            gathered.append((document, node))

    logger.info("[%s] Gathered %d raw nodes", hop_label, len(gathered))
    return gathered


def _merge_unique(
    existing: list[tuple[Document, TreeNode]],
    new: list[tuple[Document, TreeNode]],
) -> list[tuple[Document, TreeNode]]:
    seen_node_ids = {node.id for _, node in existing}
    merged = list(existing)
    for document, node in new:
        if node.id not in seen_node_ids:
            seen_node_ids.add(node.id)
            merged.append((document, node))
    return merged


async def _check_completeness(question: str, context_text: str) -> tuple[bool, str | None]:
    prompt = (
        f"Question: {question}\n\nGathered excerpts:\n{context_text}\n\n"
        "Is this enough to fully answer the question?"
    )
    response = await chat_completion(prompt=prompt, system=COMPLETENESS_CHECK_SYSTEM_PROMPT)

    stripped = response.strip()
    match = re.match(r"INCOMPLETE\s*:\s*(.+)", stripped, re.IGNORECASE | re.DOTALL)
    if match:
        missing_query = match.group(1).strip()
        logger.info("Completeness check: INCOMPLETE - missing=%r", missing_query)
        return False, missing_query or None

    logger.info("Completeness check: COMPLETE (raw reply=%r)", stripped)
    return True, None


def _build_context_text(

    sources = []
    for document, node in nodes_with_documents:
        text = (node.raw_text or "").strip() or "(no direct text - see subsections)"
        context_blocks.append(

            f"(pages {node.page_start}-{node.page_end})\n"
            f"{text[:CONTEXT_CHARS_PER_NODE]}"
        )
        sources.append({
            "document_id": document.id,
            "filename": document.filename,
            "node_id": node.id,
            "title": node.title,
            "page_start": node.page_start,
            "page_end": node.page_end,
        })

async def answer_question_multi_doc(db: Session, question: str, batch_id: str) -> dict:
    """
    Answers a question by searching across every ready document in batch_id,
    with a multi-hop loop that re-searches for missing information.
    """
    max_hops = settings.max_hops
    max_context = settings.max_total_context_nodes

    logger.info("=== New multi-document question for batch %s: %r ===", batch_id, question)

    accumulated: list[tuple[Document, TreeNode]] = []
    search_query = question
    hop = 0

    for hop in range(1, max_hops + 1):
        hop_label = f"batch={batch_id} hop={hop}/{max_hops}"

        new_nodes = await _gather_context(
            db, search_query, batch_id=batch_id, hop_label=hop_label
        )
        accumulated = _merge_unique(accumulated, new_nodes)

        if not accumulated:
            logger.info("[%s] No relevant nodes found in any candidate document yet", hop_label)
            if hop == max_hops:
                break
            continue

        reranked = await rerank_nodes(question, accumulated, top_n=max_context)
        context_text, sources = _build_context_text(reranked)

        if hop == max_hops:
            logger.info("[%s] Reached MAX_HOPS - using accumulated context as-is", hop_label)
            break

        is_complete, missing_query = await _check_completeness(question, context_text)
        if is_complete or not missing_query:
            logger.info("[%s] Context judged complete - stopping hops", hop_label)
            break

        logger.info("[%s] Context incomplete - next hop will search for: %r", hop_label, missing_query)
        search_query = missing_query

    if not accumulated:
        logger.info("No relevant nodes found across any hop in batch %s", batch_id)
        answer = await chat_completion(
            prompt=f"Question: {question}\n\n(No relevant section of any document was found.)",
            system=NO_CONTEXT_SYSTEM_PROMPT,
        )
        return {"answer": answer, "sources": [], "hops_used": hop}

    reranked = await rerank_nodes(question, accumulated, top_n=max_context)
    context_text, sources = _build_context_text(reranked)

    logger.info(
        "Generating final multi-document answer from %d context nodes "
        "(batch=%s, hops_used=%d)",
        len(reranked), batch_id, hop,
    )
    final_prompt = f"Document excerpts:\n\n{context_text}\n\nQuestion: {question}"
    answer = await chat_completion(prompt=final_prompt, system=MULTI_DOC_ANSWER_SYSTEM_PROMPT)

    return {"answer": answer, "sources": sources, "hops_used": hop}

"""
Thin wrapper around the local Ollama server.
Kept dependency-free (plain httpx) so we're not locked into a framework.

async def get_embedding(text: str) -> list[float]:
    """
    Gets a vector embedding for a single piece of text.
    Prefer get_embeddings_batch() when you have multiple texts.
    """
    vectors = await get_embeddings_batch([text])
    return vectors[0]


async def get_embeddings_batch(texts: List[str]) -> List[list[float]]:
    """
    Batch embedding: prefers /api/embed (modern, supports real batching).
    Falls back to sequential /api/embeddings calls if the server returns 404
    (older Ollama versions).
    """
    if not texts:
        return []

    # Ollama can handle reasonably large batches; chunk if extremely large
    MAX_BATCH = 64
    if len(texts) > MAX_BATCH:
        results: List[list[float]] = []
        for i in range(0, len(texts), MAX_BATCH):
            chunk = texts[i : i + MAX_BATCH]
            results.extend(await get_embeddings_batch(chunk))
        return results

    start = time.monotonic()
    try:
        async with httpx.AsyncClient(timeout=settings.ollama_timeout_seconds) as client:
            # --- Prefer modern endpoint ---
            payload = {
                "model": settings.ollama_embedding_model,
                "input": texts,
            }
            resp = await client.post(
                f"{settings.ollama_base_url}/api/embed", json=payload
            )

            if resp.status_code == 404:
                # Older Ollama – fall back to legacy single-item endpoint
                logger.warning(
                    "Ollama /api/embed returned 404 – falling back to legacy "
                    "/api/embeddings (consider upgrading Ollama)"
                )
                return await _legacy_embeddings_batch(client, texts)

            resp.raise_for_status()
            data = resp.json()
    except Exception:
        elapsed = time.monotonic() - start
        logger.exception(
            "Ollama BATCH EMBED REQUEST FAILED after %.1fs (n=%d)",
            elapsed, len(texts),
        )
        raise

    elapsed = time.monotonic() - start
    embeddings = data.get("embeddings", [])
    if len(embeddings) != len(texts):
        raise ValueError(
            f"Ollama returned {len(embeddings)} embeddings for {len(texts)} inputs"
        )

    logger.info(
        "Ollama BATCH EMBED: elapsed=%.1fs n=%d total_chars=%d dim=%d",
        elapsed,
        len(texts),
        sum(len(t) for t in texts),
        len(embeddings[0]) if embeddings else 0,
    )
    return embeddings


async def _legacy_embeddings_batch(
    client: httpx.AsyncClient, texts: List[str]
) -> List[list[float]]:
    """
    Sequential fallback using the old /api/embeddings endpoint
    (expects "prompt" and returns "embedding").
    """
    results: List[list[float]] = []
    for text in texts:
        payload = {
            "model": settings.ollama_embedding_model,
            "prompt": text,
        }
        resp = await client.post(
            f"{settings.ollama_base_url}/api/embeddings", json=payload
        )
        resp.raise_for_status()
        data = resp.json()
        emb = data.get("embedding")
        if emb is None:
            raise ValueError("Legacy /api/embeddings response missing 'embedding'")
        results.append(emb)
    return results


def _strip_thinking(content: str) -> str:
    """
    Some Ollama versions don't fully honor `"think": false` for reasoning
    models like qwen3, and still emit a hidden chain-of-thought ending in
    `</think>` before the real answer. Strip everything up to and
    including the last `</think>` so callers only see the final answer.
    """
    if "</think>" in content:
        content = content.split("</think>")[-1]
    return content.strip()

"""
PDF parsing + structure detection.

Produces a nested tree of sections from a PDF path:

    {
      "page_count": int,
      "tree": [
        {
          "title": str, "summary": None, "raw_text": str,
          "page_start": int, "page_end": int,
          "children": [ ... ]
        },
        ...
      ]
    }

Improvements for complex / scanned / large PDFs:
1. OCR fallback (PyMuPDF built-in OCR) when a page has almost no extractable text.
2. Table-aware text extraction that preserves row structure where possible.
3. Character-based chunking fallback when heading detection fails or produces
   a single giant leaf (in addition to the existing page-based expansion).
4. Oversized-leaf expansion so retrieval can drill into multi-page blobs.
"""
from collections import Counter
from typing import Optional

import pymupdf as fitz

MAX_HEADING_LEVELS = 4
MAX_HEADING_WORDS = 20
BOLD_HEADING_MAX_WORDS = 12
PAGE_CHUNK_SIZE = 5
MAX_LEAF_TEXT_CHARS = 4000
# If a page yields fewer characters than this, try OCR
MIN_CHARS_PER_PAGE_FOR_TEXT = 40
# Character-based chunk size when we fall back to pure text chunking
CHAR_CHUNK_SIZE = 2500
CHAR_CHUNK_OVERLAP = 200


def _is_bold_span(font_name: str, flags: int) -> bool:
    return bool(flags & (1 << 4)) or "bold" in (font_name or "").lower()


def _extract_page_text_with_tables(page) -> str:
    """
    Prefer structured text that keeps table-like alignment.
    Falls back to plain text if find_tables is unavailable or fails.
    """
    parts: list[str] = []

    # Try table extraction first (PyMuPDF 1.23+)
    try:
        tabs = page.find_tables()
        table_areas = []
        if tabs and tabs.tables:
            for tab in tabs.tables:
                try:
                    data = tab.extract()
                    if data:
                        rows = []
                        for row in data:
                            cells = [str(c).strip() if c is not None else "" for c in row]
                            rows.append(" | ".join(cells))
                        parts.append("\n".join(rows))
                        if hasattr(tab, "bbox"):
                            table_areas.append(tab.bbox)
                except Exception:
                    continue
    except Exception:
        table_areas = []

    # Regular text blocks, skipping regions already covered by tables when possible
    try:
        blocks = page.get_text("blocks")
        for b in blocks:
            if len(b) < 5:
                continue
            x0, y0, x1, y1, text = b[0], b[1], b[2], b[3], b[4]
            if not text or not str(text).strip():
                continue
            # Skip if this block is mostly inside a detected table
            inside_table = False
            for bbox in table_areas:
                try:
                    bx0, by0, bx1, by1 = bbox
                    if x0 >= bx0 - 2 and y0 >= by0 - 2 and x1 <= bx1 + 2 and y1 <= by1 + 2:
                        inside_table = True
                        break
                except Exception:
                    pass
            if not inside_table:
                parts.append(str(text).strip())
    except Exception:
        # Ultimate fallback
        plain = page.get_text("text") or ""
        if plain.strip():
            parts.append(plain.strip())

    return "\n".join(parts).strip()


def _ocr_page(page) -> str:
    """
    Attempt OCR on a page that has little or no extractable text.
    Uses PyMuPDF's built-in OCR support when available (requires tesseract).
    Returns empty string if OCR is not available or fails.
    """
    try:
        # PyMuPDF OCR: get_textpage_ocr is available in recent versions
        tp = page.get_textpage_ocr(language="eng", dpi=150, full=True)
        text = page.get_text("text", textpage=tp) or ""
        return text.strip()
    except Exception:
        try:
            # Older / alternative path
            text = page.get_text("text") or ""
            return text.strip()
        except Exception:
            return ""


def extract_lines(file_path: str) -> tuple[int, list[dict]]:
    """
    Extract per-line metadata (text, size, bold, page) for heading detection.
    Also performs OCR fallback and table-aware extraction for content.
    """
    doc = fitz.open(file_path)
    page_count = doc.page_count
    lines: list[dict] = []
    # Also keep full page texts for character-chunk fallback later
    page_texts: dict[int, str] = {}

    for page_index in range(page_count):
        page = doc[page_index]
        page_num = page_index + 1

        # Structured / table-aware text for content
        page_text = _extract_page_text_with_tables(page)
        if len(page_text) < MIN_CHARS_PER_PAGE_FOR_TEXT:
            ocr_text = _ocr_page(page)
            if len(ocr_text) > len(page_text):
                page_text = ocr_text

        page_texts[page_num] = page_text

        # Span-level extraction for heading detection (font size / bold)
        try:
            page_dict = page.get_text("dict")
        except Exception:
            page_dict = {"blocks": []}

        for block in page_dict.get("blocks", []):
            if block.get("type") != 0:
                continue
            for line in block.get("lines", []):
                text_parts, sizes, bold_flags = [], [], []
                for span in line.get("spans", []):
                    text = span.get("text", "")
                    if not text.strip():
                        continue
                    text_parts.append(text)
                    sizes.append(span.get("size", 0.0))
                    bold_flags.append(
                        _is_bold_span(span.get("font", ""), span.get("flags", 0))
                    )

                line_text = "".join(text_parts).strip()
                if not line_text:
                    continue

                lines.append({
                    "text": line_text,
                    "size": (sum(sizes) / len(sizes)) if sizes else 0.0,
                    "bold": all(bold_flags) if bold_flags else False,
                    "page": page_num,
                })

        # If dict extraction produced no lines but we have page_text (e.g. OCR),
        # synthesize one line so page-chunk fallback still has content.
        if not any(l["page"] == page_num for l in lines) and page_text:
            for para in page_text.split("\n"):
                para = para.strip()
                if para:
                    lines.append({
                        "text": para,
                        "size": 11.0,
                        "bold": False,
                        "page": page_num,
                    })

    doc.close()
    # Stash page_texts on a module-level attribute so parse_pdf_to_tree can use them
    extract_lines._last_page_texts = page_texts  # type: ignore[attr-defined]
    return page_count, lines


def detect_headings(lines: list[dict]) -> list[dict]:
    if not lines:
        return []

    rounded_sizes = [round(l["size"]) for l in lines]
    body_size = Counter(rounded_sizes).most_common(1)[0][0]

    larger_sizes = sorted({s for s in rounded_sizes if s > body_size}, reverse=True)
    size_to_level = {size: i + 1 for i, size in enumerate(larger_sizes[:MAX_HEADING_LEVELS])}

    headings = []
    for idx, line in enumerate(lines):
        size = round(line["size"])
        word_count = len(line["text"].split())

        if size in size_to_level:
            if word_count > MAX_HEADING_WORDS:
                continue
            level = size_to_level[size]
        elif (
            not larger_sizes
            and line["bold"]
            and size >= body_size
            and word_count <= BOLD_HEADING_MAX_WORDS
            and not line["text"].endswith((".", ",", ";", ":"))
        ):
            level = 1
        else:
            continue

        headings.append({
            "line_index": idx,
            "level": level,
            "title": line["text"],
            "page": line["page"],
        })

    return headings


def _attach_direct_text(lines: list[dict], headings: list[dict]) -> None:
    for i, h in enumerate(headings):
        start = h["line_index"] + 1
        end = headings[i + 1]["line_index"] if i + 1 < len(headings) else len(lines)
        text_lines = [lines[j]["text"] for j in range(start, end)]
        h["raw_text"] = "\n".join(text_lines).strip()
        h["page_end"] = lines[end - 1]["page"] if end > start else h["page"]
        h["_own_lines"] = lines[start:end]


def _nest_headings(headings: list[dict]) -> list[dict]:
    roots: list[dict] = []
    stack: list[tuple[int, dict]] = []

    for h in headings:
        node = {
            "title": h["title"],
            "summary": None,
            "raw_text": h["raw_text"],
            "page_start": h["page"],
            "page_end": h["page_end"],
            "children": [],
            "_own_lines": h["_own_lines"],
        }
        while stack and stack[-1][0] >= h["level"]:
            stack.pop()
        if stack:
            stack[-1][1]["children"].append(node)
        else:
            roots.append(node)
        stack.append((h["level"], node))

    return roots


def _build_page_chunks(lines: list[dict], page_count: int) -> list[dict]:
    if not lines or page_count == 0:
        return []

    nodes = []
    for start_page in range(1, page_count + 1, PAGE_CHUNK_SIZE):
        end_page = min(start_page + PAGE_CHUNK_SIZE - 1, page_count)
        chunk_text = "\n".join(
            l["text"] for l in lines if start_page <= l["page"] <= end_page
        ).strip()
        title = (
            f"Pages {start_page}-{end_page}"
            if end_page > start_page
            else f"Page {start_page}"
        )
        nodes.append({
            "title": title,
            "summary": None,
            "raw_text": chunk_text,
            "page_start": start_page,
            "page_end": end_page,
            "children": [],
        })
    return nodes


def _build_char_chunks(
    text: str,
    page_start: int,
    page_end: int,
    title_prefix: str = "Section",
) -> list[dict]:
    """
    Split a long text blob into overlapping character windows.
    Used when heading detection fails or produces oversized leaves.
    """
    text = (text or "").strip()
    if not text:
        return []

    if len(text) <= CHAR_CHUNK_SIZE:
        return [{
            "title": title_prefix,
            "summary": None,
            "raw_text": text,
            "page_start": page_start,
            "page_end": page_end,
            "children": [],
        }]

    chunks = []
    start = 0
    idx = 1
    while start < len(text):
        end = min(start + CHAR_CHUNK_SIZE, len(text))
        # Prefer to break at a paragraph or sentence boundary
        if end < len(text):
            break_at = text.rfind("\n\n", start, end)
            if break_at < start + CHAR_CHUNK_SIZE // 2:
                break_at = text.rfind(". ", start, end)
            if break_at >= start + CHAR_CHUNK_SIZE // 2:
                end = break_at + 1

        piece = text[start:end].strip()
        if piece:
            chunks.append({
                "title": f"{title_prefix} (part {idx})",
                "summary": None,
                "raw_text": piece,
                "page_start": page_start,
                "page_end": page_end,
                "children": [],
            })
            idx += 1

        if end >= len(text):
            break
        start = max(end - CHAR_CHUNK_OVERLAP, start + 1)

    return chunks


def _build_page_chunks_from_lines(
    own_lines: list[dict], page_start: int, page_end: int
) -> list[dict]:
    if not own_lines or page_end < page_start:
        return []

    nodes = []
    for chunk_start in range(page_start, page_end + 1, PAGE_CHUNK_SIZE):
        chunk_end = min(chunk_start + PAGE_CHUNK_SIZE - 1, page_end)
        chunk_text = "\n".join(
            l["text"] for l in own_lines if chunk_start <= l["page"] <= chunk_end
        ).strip()
        if not chunk_text:
            continue
        title = (
            f"Pages {chunk_start}-{chunk_end}"
            if chunk_end > chunk_start
            else f"Page {chunk_start}"
        )
        # If even a page-chunk is still huge, sub-split by characters
        if len(chunk_text) > MAX_LEAF_TEXT_CHARS:
            nodes.extend(
                _build_char_chunks(chunk_text, chunk_start, chunk_end, title_prefix=title)
            )
        else:
            nodes.append({
                "title": title,
                "summary": None,
                "raw_text": chunk_text,
                "page_start": chunk_start,
                "page_end": chunk_end,
                "children": [],
            })
    return nodes


def _expand_oversized_leaves(nodes: list[dict]) -> None:
    """
    Recursively expand leaf nodes whose raw_text exceeds MAX_LEAF_TEXT_CHARS
    into page-based (or character-based) children so retrieval can drill in.
    """
    for node in nodes:
        children = node.get("children") or []
        if children:
            _expand_oversized_leaves(children)
            continue

        raw_text = node.get("raw_text") or ""
        own_lines = node.pop("_own_lines", None)

        if len(raw_text) <= MAX_LEAF_TEXT_CHARS:
            continue

        if own_lines:
            sub_chunks = _build_page_chunks_from_lines(
                own_lines, node["page_start"], node["page_end"]
            )
        else:
            sub_chunks = _build_char_chunks(
                raw_text,
                node["page_start"],
                node["page_end"],
                title_prefix=node.get("title") or "Section",
            )

        if len(sub_chunks) > 1:
            node["children"] = sub_chunks


def _strip_internal_keys(nodes: list[dict]) -> None:
    for node in nodes:
        node.pop("_own_lines", None)
        if node.get("children"):
            _strip_internal_keys(node["children"])


def parse_pdf_to_tree(file_path: str) -> dict:
    page_count, lines = extract_lines(file_path)
    headings = detect_headings(lines)

    if headings:
        _attach_direct_text(lines, headings)
        tree = _nest_headings(headings)
        _expand_oversized_leaves(tree)
        _strip_internal_keys(tree)
    else:

"""
Lightweight embedding-based reranker.

Scores collected (document, node) pairs by cosine similarity between the
question embedding and each node's title+summary embedding, keeps top_n.

Uses batched /api/embed calls so ranking 30-40 nodes is one (or a few)
HTTP requests instead of dozens of sequential ones.
"""
import logging
import math
from typing import List, Tuple

from app.models.models import Document, TreeNode
from app.services.ollama_service import get_embedding, get_embeddings_batch

logger = logging.getLogger("pageindex.reranker")

DEFAULT_TOP_N = 10


def _cosine_similarity(a: list[float], b: list[float]) -> float:
    dot = sum(x * y for x, y in zip(a, b))
    norm_a = math.sqrt(sum(x * x for x in a))
    norm_b = math.sqrt(sum(y * y for y in b))
    if norm_a == 0 or norm_b == 0:
        return 0.0
    return dot / (norm_a * norm_b)


def _node_embedding_text(node: TreeNode) -> str:
    """
    What we embed for each node - title + summary, not the full raw_text.
    Keeps embedding calls fast and consistent in size.
    """
    summary = node.summary or ""
    return f"{node.title}. {summary}".strip()


async def rerank_nodes(
    question: str,
    nodes_with_documents: List[Tuple[Document, TreeNode]],
    top_n: int = DEFAULT_TOP_N,
) -> List[Tuple[Document, TreeNode]]:
    """
    Scores each (document, node) pair by cosine similarity and returns
    the top_n highest-scoring pairs, sorted best-first.

    If there are already <= top_n pairs, returns them unchanged.
    """
    if len(nodes_with_documents) <= top_n:
        logger.info(
            "Only %d nodes collected (<= top_n=%d) - skipping reranking, keeping all",
            len(nodes_with_documents), top_n,
        )
        return nodes_with_documents

    logger.info("Reranking %d nodes down to top %d via batch embeddings", len(nodes_with_documents), top_n)

    question_embedding = await get_embedding(question)

    # Single batch call for all node texts
    node_texts = [_node_embedding_text(node) for _, node in nodes_with_documents]
    node_embeddings = await get_embeddings_batch(node_texts)

    scored = [
        (_cosine_similarity(question_embedding, node_emb), document, node)
        for (document, node), node_emb in zip(nodes_with_documents, node_embeddings)
    ]

    scored.sort(key=lambda triple: triple[0], reverse=True)
    top = scored[:top_n]

    logger.info(
        "Reranked top %d: %s",
        len(top),
        [(round(score, 3), doc.filename, node.title) for score, doc, node in top],
    )

    return [(document, node) for _, document, node in top]

"""
Phase 4: tree-search retrieval agent.

Given a question, walks a single document's PageIndex tree top-down:
  1. At the root level, select relevant sections (LLM + keyword boost).
  2. For each selected section, recurse into children in PARALLEL.
  3. Collect leaf (or fallback) raw text as context.
  4. Deduplicate by node id so the same section is never collected twice.

Improvements over sequential version:
- Sibling subtrees are traversed concurrently via asyncio.gather.
- visited_ids set prevents duplicate collection.
- Robust number parsing from LLM replies.
- Root-level selection blends LLM picks with simple keyword overlap
  so broad questions are less likely to drop a relevant branch.
"""
import asyncio
import logging
import re
from typing import Optional

from sqlalchemy.orm import Session

from app.models.models import TreeNode
from app.services.ollama_service import chat_completion

logger = logging.getLogger("pageindex.retrieval")

MAX_SELECTIONS_PER_LEVEL = 3
MAX_LEAF_NODES = 6
MAX_DEPTH = 6
CONTEXT_CHARS_PER_NODE = 3000

SELECTION_SYSTEM_PROMPT = (
    "You are navigating a document's outline to find sections relevant to a "
    "question. You are shown a numbered list of section titles and short "
    "summaries. Reply with ONLY the numbers of sections that might help answer "
    "the question, separated by commas (example: 1,3). If none of them seem "
    "relevant, reply with exactly: NONE. Do not add any explanation."
)

ANSWER_SYSTEM_PROMPT = (
    "Answer the user's question using ONLY the document excerpts provided "
    "below. If the excerpts don't fully answer the question, say what you "
    "found and note what's missing. Do not use outside knowledge. Keep the "
    "answer concise and clear."
)

NO_CONTEXT_SYSTEM_PROMPT = (
    "Answer the question honestly. If you don't have enough information to "
    "answer it, say so plainly rather than guessing."

def _build_children_map(nodes: list[TreeNode]) -> dict[str | None, list[TreeNode]]:
    children_map: dict[str | None, list[TreeNode]] = {}
    for n in nodes:
        children_map.setdefault(n.parent_id, []).append(n)
    for key in children_map:
        children_map[key].sort(key=lambda n: n.order_index)
    return children_map


def _parse_selection_numbers(response: str, n_candidates: int) -> list[int]:
    """
    Robustly extract 1-based indices from an LLM reply.
    Accepts '1,3', '1. 3', '[1, 2]', 'sections 1 and 3', etc.
    Bounds-checks and preserves first-seen order.
    """
    numbers: list[int] = []
    seen: set[int] = set()
    for token in re.findall(r"\b\d+\b", response):
        idx = int(token)
        if 1 <= idx <= n_candidates and idx not in seen:
            seen.add(idx)
            numbers.append(idx)
    return numbers


def _keyword_overlap_score(question: str, node: TreeNode) -> float:
    """
    Cheap lexical overlap between question tokens and node title+summary+keywords.
    Used as a safety net at root level so the LLM cannot completely drop
    a section that clearly shares terms with the query.
    """
    q_tokens = set(re.findall(r"[a-z0-9]{3,}", question.lower()))
    if not q_tokens:
        return 0.0

    blob_parts = [node.title or ""]
    if node.summary:
        blob_parts.append(node.summary)
    if node.keywords:
        try:
            import json
            kws = json.loads(node.keywords)
            if isinstance(kws, list):
                blob_parts.extend(str(k) for k in kws)
        except Exception:
            blob_parts.append(node.keywords)

    blob = " ".join(blob_parts).lower()
    hits = sum(1 for t in q_tokens if t in blob)
    return hits / max(len(q_tokens), 1)


async def _select_relevant(
    question: str,
    candidates: list[TreeNode],
    level_label: str,
    *,
    is_root: bool = False,
) -> list[TreeNode]:
    if not candidates:
        return []

    listing = "\n".join(
        f"{i + 1}. {n.title} - {n.summary or '(no summary yet)'}"
        for i, n in enumerate(candidates)
    )
    prompt = (
        f"Question: {question}\n\nSections:\n{listing}\n\n"
        "Which section numbers might contain the answer?"
    )

    logger.info(
        "[%s] Asking LLM to pick among %d candidates: %s",
        level_label, len(candidates), [n.title for n in candidates],
    )

    response = await chat_completion(prompt=prompt, system=SELECTION_SYSTEM_PROMPT)
    numbers = _parse_selection_numbers(response, len(candidates))
    selected = [candidates[n - 1] for n in numbers]
    selected = selected[:MAX_SELECTIONS_PER_LEVEL]

    # Root-level safety net: if LLM returned few/none, boost by keyword overlap
    if is_root and len(selected) < MAX_SELECTIONS_PER_LEVEL:
        scored = sorted(
            (( _keyword_overlap_score(question, n), n) for n in candidates),
            key=lambda x: x[0],
            reverse=True,
        )
        already = {n.id for n in selected}
        for score, node in scored:
            if score < 0.15:
                break
            if node.id not in already:
                selected.append(node)
                already.add(node.id)
                if len(selected) >= MAX_SELECTIONS_PER_LEVEL:
                    break
        if scored and scored[0][0] > 0:
            logger.info(
                "[%s] Keyword boost added sections: %s",
                level_label,
                [n.title for n in selected if n.id not in {candidates[i-1].id for i in numbers}],
            )

    logger.info(
        "[%s] LLM raw reply=%r -> parsed_numbers=%s -> selected=%s",
        level_label, response, numbers, [n.title for n in selected],
    )
    return selected


async def _walk(
    question: str,
    node: TreeNode,
    children_map: dict[str | None, list[TreeNode]],
    depth: int,
    collected: list[TreeNode],
    visited_ids: set[str],
    lock: asyncio.Lock,
) -> None:
    async with lock:
        if len(collected) >= MAX_LEAF_NODES or depth > MAX_DEPTH or node.id in visited_ids:
            return
        visited_ids.add(node.id)

    children = children_map.get(node.id, [])
    logger.info("[depth=%d] Visiting node %r (%d children)", depth, node.title, len(children))

    if not children:
        async with lock:
            if len(collected) < MAX_LEAF_NODES and node.id not in {n.id for n in collected}:
                logger.info("[depth=%d] Leaf node - collecting %r", depth, node.title)
                collected.append(node)
        return

    selected_children = await _select_relevant(
        question, children, level_label=f"depth={depth}"
    )

    if not selected_children:
        async with lock:
            if (
                node.raw_text
                and node.raw_text.strip()
                and len(collected) < MAX_LEAF_NODES
                and node.id not in {n.id for n in collected}
            ):
                logger.info(
                    "[depth=%d] No relevant children - falling back to own text of %r",
                    depth, node.title,
                )
                collected.append(node)
            else:
                logger.info(
                    "[depth=%d] No relevant children and no own text for %r - dropping branch",
                    depth, node.title,
                )
        return

    # Parallel traversal of sibling branches
    tasks = [
        _walk(question, child, children_map, depth + 1, collected, visited_ids, lock)
        for child in selected_children
    ]
    await asyncio.gather(*tasks)


async def collect_relevant_nodes(
    db: Session,
    document_id: str,
    question: str,
    nodes: list[TreeNode] | None = None,
) -> list[TreeNode]:
    """
    Runs the tree-walk against ONE document and returns the collected
    TreeNode rows relevant to `question` - no LLM synthesis.
    """
    if nodes is None:
        nodes = db.query(TreeNode).filter(TreeNode.document_id == document_id).all()

    if not nodes:
        logger.info("Document %s has no tree nodes - nothing to collect", document_id)
        return []

    children_map = _build_children_map(nodes)
    roots = children_map.get(None, [])

    selected_roots = await _select_relevant(
        question, roots, level_label=f"doc={document_id} root", is_root=True
    )

    collected: list[TreeNode] = []
    visited_ids: set[str] = set()
    lock = asyncio.Lock()

    # Parallel walk from each selected root
    tasks = [
        _walk(question, root, children_map, depth=1, collected=collected,
              visited_ids=visited_ids, lock=lock)
        for root in selected_roots
    ]
    await asyncio.gather(*tasks)

    # Final dedup (defensive)
    seen = set()
    unique: list[TreeNode] = []
    for n in collected:
        if n.id not in seen:
            seen.add(n.id)
            unique.append(n)

    logger.info(
        "Document %s: collected %d node(s) for question %r: %s",
        document_id, len(unique), question, [n.title for n in unique],
    )
    return unique

def _parse_summary_response(response: str) -> tuple[str, list[str]]:

    summary = ""
    keywords: list[str] = []
    for line in response.splitlines():
        line = line.strip()
        if not line:

        lower = line.lower()
        if lower.startswith("summary"):
            _, _, rest = line.partition(":")
            summary = rest.strip()
        elif lower.startswith("keyword"):
            _, _, rest = line.partition(":")
            raw = rest.strip()

            parts = re.split(r"[;,\n]", raw)
            keywords = [k for p in parts if (k := _clean_keyword(p))]
    if not summary:

    # Deduplicate while preserving order
    seen = set()
    unique_kw = []
    for k in keywords:
        kl = k.lower()
        if kl not in seen:
            seen.add(kl)
            unique_kw.append(k)
    return summary, unique_kw[:12]

            order.extend(post_order(child.id))
            order.append(child.id)
        return order

    ordered_ids = post_order(None)

    # Process level-by-level in post-order batches of independent nodes.
    # Nodes whose children are already summarized can run in parallel.
    # We still respect post-order: a parent only runs after all its children.
    remaining = set(ordered_ids)
    completed: set[str] = set()

    while remaining:
        # A node is ready when all its children (that are in the tree) are done
        ready = []
        for nid in list(remaining):
            child_ids = [c.id for c in children_map.get(nid, [])]
            if all(cid in completed for cid in child_ids):
                ready.append(nid)

        if not ready:
            # Safety: if dependency cycle somehow exists, finish remaining sequentially
            ready = list(remaining)[: settings.max_concurrent_summaries]

        async def _run_one(nid: str) -> tuple[str, dict]:
            node = by_id[nid]
            child_summaries = [
                summaries[c.id]
                for c in children_map.get(nid, [])
                if c.id in summaries
            ]
            result = await _summarize_node(node, child_summaries, semaphore)
            return nid, result

        results = await asyncio.gather(*[_run_one(nid) for nid in ready])

        for nid, result in results:
            summaries[nid] = result["summary"]
            node = by_id[nid]
            node.summary = result["summary"]
            node.keywords = json.dumps(result["keywords"])
            db.add(node)
            completed.add(nid)
            remaining.discard(nid)

        db.commit()
        logger.info(
            "Document %s: summarized %d/%d nodes",
            document_id, len(completed), len(ordered_ids),
        )


async def _rollup_batch(section_listings: list[str]) -> tuple[str, list[str]]:
    """Run one rollup LLM call over a list of 'title: summary' lines."""
    listing = "\n".join(section_listings)
    prompt = f"Section summaries:\n{listing}"
    response = await chat_completion(prompt=prompt, system=DOC_ROLLUP_SYSTEM_PROMPT)
    return _parse_summary_response(response)


async def _generate_document_rollup(db: Session, document_id: str) -> None:
    roots = (
        db.query(TreeNode)
        .filter(TreeNode.document_id == document_id, TreeNode.parent_id.is_(None))
        .order_by(TreeNode.order_index)
        .all()
    )
    if not roots:
        return

    listings = [
        f"- {n.title}: {n.summary or '(no summary)'}"
        for n in roots
    ]

    if len(listings) <= MAX_ROOTS_FOR_ROLLUP:
        summary, keywords = await _rollup_batch(listings)
    else:
        # Sub-batch: roll up groups of ROLLUP_BATCH_SIZE, then roll up the intermediates
        logger.info(
            "Document %s has %d roots (> %d) - using sub-batched rollup",
            document_id, len(listings), MAX_ROOTS_FOR_ROLLUP,
        )
        intermediate: list[str] = []
        for i in range(0, len(listings), ROLLUP_BATCH_SIZE):
            chunk = listings[i : i + ROLLUP_BATCH_SIZE]
            s, _ = await _rollup_batch(chunk)
            intermediate.append(f"- Part {i // ROLLUP_BATCH_SIZE + 1}: {s}")

        summary, keywords = await _rollup_batch(intermediate)

    document = db.query(Document).filter(Document.id == document_id).first()
    if document:
        document.summary = summary or document.filename
        document.keywords = json.dumps(keywords)
        db.add(document)
        db.commit()
        logger.info(
            "Document %s rollup: summary=%r keywords=%s",
            document_id, summary, keywords,
        )

    document-level summary+keywords."""

        await _summarize_tree_async(db, document_id)

"""
Persists the nested tree produced by pdf_service.parse_pdf_to_tree()
into the `tree_nodes` table, preserving parent/child relationships and
sibling order.

Performance: generates UUIDs on the application side and prepares all
TreeNode objects in memory, then adds them in one pass so SQLAlchemy
can batch the INSERTs. This avoids the previous per-node db.flush()
which issued one SQL round-trip per node on large trees.
"""
import re
import uuid
from typing import Optional

from sqlalchemy.orm import Session

from app.models.models import TreeNode

MAX_TITLE_LENGTH = 500


def _normalize_text(text: Optional[str]) -> Optional[str]:
    """
    Minimal whitespace cleanup before storing raw_text:
    - Collapse runs of 3+ newlines into a double newline (paragraph break)
    - Collapse runs of spaces/tabs into a single space
    - Strip leading/trailing whitespace
    Does NOT destroy meaningful structure (tables, lists).
    """
    if not text:
        return text
    # Normalize Windows/Mac newlines
    text = text.replace("\r\n", "\n").replace("\r", "\n")
    # Collapse excessive blank lines
    text = re.sub(r"\n{3,}", "\n\n", text)
    # Collapse horizontal whitespace runs (but keep single spaces)
    text = re.sub(r"[ \t]+", " ", text)
    # Clean spaces around newlines
    text = re.sub(r" *\n *", "\n", text)
    return text.strip()


def _prepare_nodes(
    nodes: list[dict],
    document_id: str,
    parent_id: Optional[str] = None,
) -> list[TreeNode]:
    """
    Recursively walk the in-memory tree dicts and build TreeNode ORM
    objects with pre-generated UUIDs so parent_id links are known
    before any DB write.
    """
    prepared: list[TreeNode] = []

    for order_index, node in enumerate(nodes):
        node_id = str(uuid.uuid4())
        title = (node.get("title") or "Untitled")[:MAX_TITLE_LENGTH]
        raw_text = _normalize_text(node.get("raw_text"))

        db_node = TreeNode(
            id=node_id,
            document_id=document_id,
            parent_id=parent_id,
            title=title,
            summary=node.get("summary"),
            raw_text=raw_text,
            page_start=node.get("page_start"),
            page_end=node.get("page_end"),
            order_index=order_index,
        )
        prepared.append(db_node)

        children = node.get("children") or []
        if children:
            prepared.extend(
                _prepare_nodes(children, document_id, parent_id=node_id)
            )

    return prepared


def persist_tree(
    db: Session,
    document_id: str,
    nodes: list[dict],
    parent_id: Optional[str] = None,
) -> None:
    """
    Convert the nested tree into TreeNode rows and insert them in a
    single bulk operation (one commit by the caller is enough).
    """
    all_nodes = _prepare_nodes(nodes, document_id, parent_id=parent_id)
    if not all_nodes:
        return

    # add_all lets SQLAlchemy batch the inserts efficiently
    db.add_all(all_nodes)
    db.flush()  # one flush for the whole tree, not per node

  return (
    <div className="card">
      <h2>Documents</h2>

import React, { useState } from "react";
import { useNavigate } from "react-router-dom";
import { uploadBatch } from "../api/client.js";

export default function UploadPage() {
  const [files, setFiles] = useState([]);
  const [uploading, setUploading] = useState(false);
  const [error, setError] = useState(null);
  const navigate = useNavigate();

  const handleFileChange = (e) => {
    setFiles(Array.from(e.target.files || []));
    setError(null);
  };

  const handleUpload = async () => {
    if (files.length === 0) return;
    setUploading(true);
    setError(null);
    try {
      await uploadBatch(files);
      navigate("/batches");
    } catch (err) {
      setError(
        err?.response?.data?.detail || "Upload failed. Please try again.",
      );
    } finally {
      setUploading(false);
    }
  };

  return (
    <div className="card">
      <h2>Upload documents</h2>
      <p style={{ color: "var(--text-dim)" }}>
        Select one or more PDFs. Files selected together become one batch -
        they'll be indexed one at a time, then you can chat across all of them
        at once.
      </p>

      <input
        type="file"
        accept="application/pdf"
        multiple
        onChange={handleFileChange}
        disabled={uploading}
      />

      {files.length > 0 && (
        <ul style={{ marginTop: 12 }}>
          {files.map((f, i) => (
            <li key={i} style={{ fontSize: 14 }}>
              {f.name}
            </li>
          ))}
        </ul>
      )}

      {error && (
        <p style={{ color: "var(--danger, #e5484d)", marginTop: 12 }}>
          {error}
        </p>
      )}

      <div style={{ marginTop: 16 }}>
        <button
          onClick={handleUpload}
          disabled={uploading || files.length === 0}
        >
          {uploading
            ? "Uploading..."
            : `Upload ${files.length || ""} document${files.length === 1 ? "" : "s"}`}
        </button>
      </div>
    </div>
  );
}
